# Cue App: Communications Center

## Feature Implementation Plan & Deliverables

---

### Overview

This document delivers the complete Communications Center feature for the Cue App — a unified workspace for event-related communications including invitations, RSVP reminders, event updates, thank-you emails, and post-event surveys. All data is local/mock for Sprint 2 with no real email provider integration.

---

## Architecture Decisions

### Data Layer
- **Local mock store** using Zustand with localStorage persistence
- **No external API calls** — all AI drafting uses template-based generation with placeholder substitution
- **TypeScript-first** with strict interfaces for all communication entities

## Core Type Definitions

**`src/features/communications/types.ts`**

```typescript
// ─── Enums ───────────────────────────────────────────────────────────────────

export type CommunicationType =
  | 'invitation'
  | 'rsvp_reminder'
  | 'event_update'
  | 'thank_you'
  | 'post_event_survey';

export type CommunicationStatus =
  | 'draft'
  | 'scheduled'
  | 'sending'
  | 'sent'
  | 'failed'
  | 'cancelled';

export type AudienceSegmentType =
  | 'all_invitees'
  | 'rsvp_yes'
  | 'rsvp_no'
  | 'rsvp_pending'
  | 'attended'
  | 'did_not_attend'
  | 'custom';

export type TemplateCategory = CommunicationType | 'general';

// ─── Audience ─────────────────────────────────────────────────────────────────

export interface Recipient {
  id: string;
  name: string;
  email: string;
  rsvpStatus: 'yes' | 'no' | 'pending' | 'waitlisted';
  attended?: boolean;
  tags: string[];
}

export interface AudienceSegment {
  id: string;
  name: string;
  type: AudienceSegmentType;
  filters: SegmentFilter[];
  recipientIds: string[];          // resolved at send time
  recipientCount: number;
}

export interface SegmentFilter {
  field: keyof Recipient | string;
  operator: 'equals' | 'not_equals' | 'contains' | 'in' | 'not_in';
  value: string | string[] | boolean;
}

// ─── Templates ────────────────────────────────────────────────────────────────

export interface CommunicationTemplate {
  id: string;
  name: string;
  category: TemplateCategory;
  subject: string;
  bodyHtml: string;
  bodyText: string;
  variables: TemplateVariable[];   // e.g. {{event_name}}, {{recipient_name}}
  isBuiltIn: boolean;
  createdAt: string;
  updatedAt: string;
}

export interface TemplateVariable {
  key: string;                     // e.g. "event_name"
  label: string;                   // e.g. "Event Name"
  defaultValue: string;
  required: boolean;
}

// ─── Campaigns ────────────────────────────────────────────────────────────────

export interface Campaign {
  id: string;
  eventId: string;
  name: string;
  type: CommunicationType;
  status: CommunicationStatus;
  subject: string;
  bodyHtml: string;
  bodyText: string;
  templateId?: string;
  audienceSegment: AudienceSegment;
  scheduledAt?: string;            // ISO 8601 — null = send immediately
  sentAt?: string;
  createdAt: string;
  updatedAt: string;
  stats: CampaignStats;
  aiDraftMetadata?: AIDraftMetadata;
}

export interface CampaignStats {
  totalRecipients: number;
  sent: number;
  delivered: number;
  opened: number;
  clicked: number;
  bounced: number;
  unsubscribed: number;
}

export interface AIDraftMetadata {
  generatedAt: string;
  promptType: CommunicationType;
  toneUsed: DraftTone;
  variablesUsed: string[];
  wasEdited: boolean;
}

// ─── AI Drafting ──────────────────────────────────────────────────────────────

export type DraftTone = 'formal' | 'friendly' | 'casual' | 'urgent' | 'warm';

export interface AIDraftRequest {
  eventId: string;
  communicationType: CommunicationType;
  tone: DraftTone;
  eventContext: EventContext;
  additionalInstructions?: string;
}

export interface AIDraftResult {
  subject: string;
  bodyHtml: string;
  bodyText: string;
  suggestedSegment: AudienceSegmentType;
  metadata: AIDraftMetadata;
}

export interface EventContext {
  eventName: string;
  eventDate: string;
  eventLocation: string;
  organizerName: string;
  rsvpDeadline?: string;
  surveyUrl?: string;
}

// ─── History ──────────────────────────────────────────────────────────────────

export interface CommunicationHistoryEntry {
  id: string;
  campaignId: string;
  eventId: string;
  type: CommunicationType;
  subject: string;
  sentTo: string[];                // recipient IDs
  sentAt: string;
  status: 'delivered' | 'bounced' | 'failed';
  stats: CampaignStats;
}
```
