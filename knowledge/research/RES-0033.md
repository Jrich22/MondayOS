### 3.2 State Hook — `src/hooks/useAIAssistant.ts`

```typescript
import { useState, useCallback } from "react";

export type AIAssistantState = "closed" | "open";

interface UseAIAssistantReturn {
  isOpen: boolean;
  open: () => void;
  close: () => void;
  toggle: () => void;
}

/**
 * Manages open/closed state for the AI Assistant panel.
 * No real AI calls — placeholder state only.
 */
export function useAIAssistant(): UseAIAssistantReturn {
  const [state, setState] = useState<AIAssistantState>("closed");

  const open = useCallback(() => setState("open"), []);
  const close = useCallback(() => setState("closed"), []);
  const toggle = useCallback(
    () => setState((s) => (s === "open" ? "closed" : "open")),
    []
  );

  return {
    isOpen: state === "open",
    open,
    close,
    toggle,
  };
}
```

### 3.4 Main Panel — `src/components/ai/AIAssistantPanel.tsx`

```typescript
import React, { useEffect, useRef } from "react";
import { AI_ASSISTANT_COPY, AI_ASSISTANT_TEST_IDS } from "../../constants/ai";

interface AIAssistantPanelProps {
  isOpen: boolean;
  onClose: () => void;
}

/**
 * AI Assistant placeholder panel.
 *
 * Renders a dismissible surface explaining that AI planning features
 * are coming soon. No real AI calls are made.
 *
 * Accessibility:
 * - role="dialog" with aria-modal and aria-labelledby
 * - Focus is trapped to the panel when open
 * - Escape key closes the panel
 * - Focus returns to trigger on close (handled by parent via ref if needed)
 */
export const AIAssistantPanel: React.FC<AIAssistantPanelProps> = ({
  isOpen,
  onClose,
}) => {
  const dismissRef = useRef<HTMLButtonElement>(null);

  // Focus dismiss button when panel opens
  useEffect(() => {
    if (isOpen && dismissRef.current) {
      dismissRef.current.focus();
    }
  }, [isOpen]);

  // Close on Escape
  useEffect(() => {
    if (!isOpen) return;
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === "Escape") {
        onClose();
      }
    };
    document.addEventListener("keydown", handleKeyDown);
    return () => document.removeEventListener("keydown", handleKeyDown);
  }, [isOpen, onClose]);

  if (!isOpen) return null;

  return (
    <div
      role="dialog"
      aria-modal="true"
      aria-labelledby={AI_ASSISTANT_TEST_IDS.heading}
      data-testid={AI_ASSISTANT_TEST_IDS.panel}
      className="ai-assistant-panel"
    >
      {/* Header */}
      <div className="ai-assistant-panel__header">
        <h2
          id={AI_ASSISTANT_TEST_IDS.heading}
          data-testid={AI_ASSISTANT_TEST_IDS.heading}
          className="ai-assistant-panel__title"
        >
          {AI_ASSISTANT_COPY.panelTitle}
        </h2>
        <button
          ref={dismissRef}
          type="button"
          data-testid={AI_ASSISTANT_TEST_IDS.dismissButton}
          aria-label={AI_ASSISTANT_COPY.dismissLabel}
          onClick={onClose}
          className="ai-assistant-panel__dismiss"
        >
          ✕
        </button>
      </div>

      {/* Body */}
      <div className="ai-assistant-panel__body">
        {/* Coming Soon Badge */}
        <span
          data-testid={AI_ASSISTANT_TEST_IDS.badge}
          className="ai-assistant-panel__badge"
          aria-label={AI_ASSISTANT_COPY.comingSoonBadge}
