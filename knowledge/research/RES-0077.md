import { EventEmitter } from 'events';

// ─── Types ────────────────────────────────────────────────────────────────────

export type LifecycleState =
  | 'planning'
  | 'inviting'
  | 'live'
  | 'follow_up'
  | 'insights';

export type LifecycleTransitionTrigger =
  | 'manual_override'
  | 'date_threshold'
  | 'attendee_threshold'
  | 'event_start'
  | 'event_end'
  | 'post_event_window_close';

export interface LifecycleTransition {
  from: LifecycleState;
  to: LifecycleState;
  trigger: LifecycleTransitionTrigger;
  timestamp: Date;
  actorId?: string; // present on manual overrides
  metadata?: Record<string, unknown>;
}

export interface EventLifecycleContext {
  eventId: string;
  state: LifecycleState;
  eventStartAt: Date;
  eventEndAt: Date;
  attendeeCount: number;
  rsvpCount: number;
  history: LifecycleTransition[];
  isManualOverride: boolean;
  lastEvaluatedAt: Date;
}

// ─── Transition Rules ─────────────────────────────────────────────────────────

/**
 * Ordered set of automatic transition rules.
 * Rules are evaluated in sequence; first match wins.
 */
const TRANSITION_RULES: Array<{
  from: LifecycleState;
  to: LifecycleState;
  trigger: LifecycleTransitionTrigger;
  condition: (ctx: EventLifecycleContext, now: Date) => boolean;
}> = [
  {
    from: 'planning',
    to: 'inviting',
    trigger: 'attendee_threshold',
    // Move to Inviting once at least one RSVP link has been sent (rsvpCount > 0)
    // OR when we're 30+ days out and the event has a venue set (indicated by attendeeCount seed)
    condition: (ctx, now) =>
      ctx.rsvpCount > 0 ||
      msUntil(ctx.eventStartAt, now) <= days(30),
  },
  {
    from: 'inviting',
    to: 'live',
    trigger: 'event_start',
    condition: (ctx, now) => now >= ctx.eventStartAt,
  },
  {
    from: 'live',
    to: 'follow_up',
    trigger: 'event_end',
    condition: (ctx, now) => now >= ctx.eventEndAt,
  },
  {
    from: 'follow_up',
    to: 'insights',
    trigger: 'post_event_window_close',
    // Follow-up window = 7 days post-event
    condition: (ctx, now) =>
      msUntil(ctx.eventEndAt, now) <= -days(7),
  },
];

// Allow manual override to any valid state from any state
const MANUAL_OVERRIDE_ALLOWED: Record<LifecycleState, LifecycleState[]> = {
  planning:  ['inviting', 'live', 'follow_up', 'insights'],
  inviting:  ['planning', 'live', 'follow_up', 'insights'],
  live:      ['planning', 'inviting', 'follow_up', 'insights'],
  follow_up: ['planning', 'inviting', 'live', 'insights'],
  insights:  ['planning', 'inviting', 'live', 'follow_up'],
};

// ─── Engine ───────────────────────────────────────────────────────────────────

export class LifecycleEngine extends EventEmitter {
  private contexts = new Map<string, EventLifecycleContext>();

  /**
   * Initialise a new event context. Idempotent — safe to call on app boot.
   */
  initialise(
    eventId: string,
    params: Pick<
      EventLifecycleContext,
      'eventStartAt' | 'eventEndAt' | 'attendeeCount' | 'rsvpCount'
    >,
    initialState: LifecycleState = 'planning',
  ): EventLifecycleContext {
    if (this.contexts.has(eventId)) {
      return this.contexts.get(eventId)!;
    }
    const ctx: EventLifecycleContext = {
      eventId,
      state: initialState,
      ...params,
      history: [],
      isManualOverride: false,
      lastEvaluatedAt: new Date(),
    };
    this.contexts.set(eventId, ctx);
    return ctx;
  }

  /**
   * Evaluate automatic transitions for a given event.
   * Should be called on a scheduler tick (e.g. every 5 min).
   */
  evaluate(eventId: string, now = new Date()): LifecycleTransition | null {
    const ctx = this.requireContext(eventId);

    // Manual override locks automatic evaluation until cleared
    if (ctx.isManualOverride) return null;

    ctx.lastEvaluatedAt = now;

    const rule = TRANSITION_RULES.find(
      (r) => r.from === ctx.state && r.condition(ctx, now),
    );

    if (!rule) return null;

    return this.applyTransition(ctx, rule.to, rule.trigger, now);
  }

  /**
   * Force a state transition. Records the actorId for audit trail.
   */
  manualOverride(
    eventId: string,
    targetState: LifecycleState,
    actorId: string,
    metadata?: Record<string, unknown>,
  ): LifecycleTransition {
    const ctx = this.requireContext(eventId);
    const allowed = MANUAL_OVERRIDE_ALLOWED[ctx.state];

    if (!allowed.includes(targetState)) {
      throw new Error(
        `Transition from '${ctx.state}' to '${targetState}' is not permitted.`,
      );
    }

    ctx.isManualOverride = true;
    return this.applyTransition(ctx, targetState, 'manual_override', new Date(), actorId, metadata);
  }

  /**
   * Resume automatic transitions (clears manual override lock).
   */
  resumeAutoTransitions(eventId: string): void {
    this.requireContext(eventId).isManualOverride = false;
  }

  getContext(eventId: string): EventLifecycleContext | undefined {
    return this.contexts.get(eventId);
  }

  // ── Private ────────────────────────────────────────────────────────────────

  private applyTransition(
    ctx: EventLifecycleContext,
    to: LifecycleState,
    trigger: LifecycleTransitionTrigger,
    now: Date,
    actorId?: string,
    metadata?: Record<string, unknown>,
  ): LifecycleTransition {
    const transition: LifecycleTransition = {
      from: ctx.state,
      to,
      trigger,
      timestamp: now,
      actorId,
      metadata,
    };

    ctx.state = to;

    this.emit(`state:${to}`, { eventId: ctx.eventId, context: ctx });

    return transition;
  }
