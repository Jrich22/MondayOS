### 3.3 `EventCard.tsx`

```tsx
// src/components/EventDashboard/EventCard.tsx
import React from 'react';
import { CueEvent } from '../../types/event';
import { EventStatusBadge } from './EventStatusBadge';
import styles from './EventDashboard.module.css';

interface Props {
  event: CueEvent;
  onSelect: (id: string) => void;
}

function formatDate(iso: string): string {
  return new Intl.DateTimeFormat('en-GB', {
    day: 'numeric',
    month: 'short',
    year: 'numeric',
    hour: '2-digit',
    minute: '2-digit',
  }).format(new Date(iso));
}

function capacityLabel(sold: number, capacity: number): string {
  if (capacity === 0) return '—';
  const pct = Math.round((sold / capacity) * 100);
  return `${sold} / ${capacity} (${pct}%)`;
}

export const EventCard: React.FC<Props> = ({ event, onSelect }) => {
  const handleClick = () => onSelect(event.id);
  const handleKey = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter' || e.key === ' ') onSelect(event.id);
  };

  return (
    <li
      className={styles.card}
      onClick={handleClick}
      onKeyDown={handleKey}
      tabIndex={0}
      role="button"
      aria-label={`View event: ${event.title}`}
      data-testid={`event-card-${event.id}`}
    >
      <div className={styles.card_main}>
        <span className={styles.card_title}>{event.title}</span>
        <EventStatusBadge status={event.status} />
      </div>

      <dl className={styles.card_meta}>
        <div>
          <dt className={styles.sr_only}>Date</dt>
          <dd>{formatDate(event.date)}</dd>
        </div>
        <div>
          <dt className={styles.sr_only}>Venue</dt>
          <dd>{event.venue}</dd>
        </div>
        <div>
          <dt className={styles.sr_only}>Capacity</dt>
          <dd>{capacityLabel(event.ticketsSold, event.capacity)}</dd>
        </div>
      </dl>
    </li>
  );
};
```
