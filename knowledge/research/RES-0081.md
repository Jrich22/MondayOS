# Cue App: Mission Control Dashboard

## Engineering Delivery Document

---

## Executive Summary

This document delivers a concrete, actionable implementation plan and working code for the **Mission Control Dashboard** — the organizer's operational home screen in the Cue App. The implementation covers live event health monitoring, attendance tracking, AI recommendations, milestone tracking, and organizational metrics, architected to scale from single-event → organization-wide → portfolio-wide views.

---

## 1. Architecture Overview

```
┌─────────────────────────────────────────────────────────────┐
│                    Mission Control Dashboard                  │
├─────────────────┬──────────────────┬────────────────────────┤
│   Data Layer    │  State Layer      │   Presentation Layer   │
│                 │                  │                        │
│ • EventStream   │ • DashboardStore  │ • MissionControl       │
│ • MetricsAPI    │ • AlertsStore     │   (root component)     │
│ • AIEngine      │ • UIStore         │ • EventHealthCard      │
│ • WebSocket     │                  │ • AttendancePanel      │
│   Gateway       │                  │ • AlertsFeed           │
│                 │                  │ • AIRecommendations    │
│                 │                  │ • MilestoneTracker     │
│                 │                  │ • OrgMetrics           │
└─────────────────┴──────────────────┴────────────────────────┘

Scope Evolution:
  Phase 1: Single Event  →  scope=event&id={eventId}
  Phase 2: Org-wide      →  scope=organization&id={orgId}
  Phase 3: Portfolio     →  scope=portfolio&id={portfolioId}
```

// src/types/mission-control.ts

export type DashboardScope = 'event' | 'organization' | 'portfolio';

export type HealthStatus = 'healthy' | 'warning' | 'critical' | 'unknown';

export type AlertSeverity = 'info' | 'warning' | 'critical';

export interface EventHealth {
  eventId: string;
  eventName: string;
  startTime: Date;
  endTime: Date;
  venue: string;
  status: HealthStatus;
  healthScore: number; // 0-100
  indicators: HealthIndicator[];
  lastUpdated: Date;
}

export interface HealthIndicator {
  id: string;
  label: string;
  status: HealthStatus;
  value: string | number;
  threshold?: number;
  trend: 'up' | 'down' | 'stable';
  detail?: string;
}

export interface AttendanceMetrics {
  eventId: string;
  registered: number;
  checkedIn: number;
  checkedOut: number;
  capacity: number;
  checkInRate: number; // percentage
  peakOccupancy: number;
  currentOccupancy: number;
  projectedTotal: number;
  checkInVelocity: number; // per minute
  checkInHistory: TimeSeriesPoint[];
}

export interface TimeSeriesPoint {
  timestamp: Date;
  value: number;
}

export interface Alert {
  id: string;
  eventId?: string;
  orgId?: string;
  severity: AlertSeverity;
  category: AlertCategory;
  title: string;
  description: string;
  timestamp: Date;
  acknowledged: boolean;
  acknowledgedBy?: string;
  acknowledgedAt?: Date;
  actionUrl?: string;
  actionLabel?: string;
  autoResolveAt?: Date;
}

export type AlertCategory =
  | 'attendance'
  | 'capacity'
  | 'technical'
  | 'financial'
  | 'vendor'
  | 'safety'
  | 'ai_insight';

export interface AIRecommendation {
  id: string;
  eventId?: string;
  orgId?: string;
  priority: 'high' | 'medium' | 'low';
  category: string;
  headline: string;
  rationale: string;
  suggestedAction: string;
  estimatedImpact: string;
  confidence: number; // 0-1
  generatedAt: Date;
  expiresAt?: Date;
  dismissed: boolean;
}

export interface Milestone {
  id: string;
  eventId: string;
  title: string;
  dueDate: Date;
  completedAt?: Date;
  status: 'pending' | 'in_progress' | 'completed' | 'overdue' | 'at_risk';
  owner: string;
  category: MilestoneCategory;
  blockedBy?: string[];
}

export type MilestoneCategory =
  | 'venue'
  | 'marketing'
  | 'logistics'
  | 'technology'
  | 'staffing'
  | 'compliance';

export interface OrgMetrics {
  orgId: string;
  period: 'day' | 'week' | 'month' | 'quarter' | 'year';
  totalEvents: number;
  activeEvents: number;
  upcomingEvents: number;
  totalAttendees: number;
  averageHealthScore: number;
  revenueTotal: number;
  revenueTrend: number; // percentage change
  npsScore: number;
  topEvents: EventSummary[];
}

export interface EventSummary {
  eventId: string;
  eventName: string;
  date: Date;
  attendees: number;
  healthScore: number;
  revenue?: number;
}

export interface DashboardConfig {
  scope: DashboardScope;
  scopeId: string;
  refreshInterval: number; // ms
  alertsEnabled: boolean;
  aiEnabled: boolean;
  panels: PanelConfig[];
}

export interface PanelConfig {
  id: string;
  type: PanelType;
  visible: boolean;
  position: { row: number; col: number; rowSpan: number; colSpan: number };
}

export type PanelType =
  | 'event_health'
  | 'attendance'
  | 'alerts'
  | 'ai_recommendations'
  | 'milestones'
  | 'org_metrics'
  | 'event_list';
