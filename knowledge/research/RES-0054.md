# Create-Event Flow — Implementation

## Overview

This implementation delivers the complete create-event flow for the `cue-app` component, producing a persisted event record with its associated workspace. I'll provide concrete, production-ready code aligned with MVP scope constraints.

---

## Architecture Decision

Based on the repository context (TASK-0020 `event_workspace`, depends on TASK-0030), the flow follows:

```
User Input → Validation → Event Record Creation → Workspace Initialization → Persistence → Confirmation
```

```typescript
export const EventStatusSchema = z.enum(['draft', 'published', 'archived']);
export type EventStatus = z.infer<typeof EventStatusSchema>;

export const CreateEventInputSchema = z.object({
  title: z
    .string()
    .min(3, 'Title must be at least 3 characters')
    .max(120, 'Title must be at most 120 characters')
    .trim(),
  description: z
    .string()
    .max(2000, 'Description must be at most 2000 characters')
    .trim()
    .optional(),
  startsAt: z
    .string()
    .datetime({ message: 'startsAt must be a valid ISO-8601 datetime' }),
  endsAt: z
    .string()
    .datetime({ message: 'endsAt must be a valid ISO-8601 datetime' }),
  timezone: z
    .string()
    .min(1, 'Timezone is required')
    .default('UTC'),
  capacity: z
    .number()
    .int()
    .positive()
    .max(100_000)
    .optional(),
  isVirtual: z.boolean().default(false),
  location: z
    .string()
    .max(255)
    .trim()
    .optional(),
}).refine(
  (data) => new Date(data.endsAt) > new Date(data.startsAt),
  {
    message: 'endsAt must be after startsAt',
    path: ['endsAt'],
  }
);

export type CreateEventInput = z.infer<typeof CreateEventInputSchema>;

  id: string;
  workspaceId: string;
  ownerId: string;
  title: string;
  description?: string;
  startsAt: Date;
  endsAt: Date;
  timezone: string;

  location?: string;
  status: EventStatus;

}
```

### 2. Repository Layer

**`src/repositories/event.repository.ts`**
```typescript
import { db } from '../db/client';
import { Event, CreateEventInput } from '../models/event';
import { Workspace } from '../models/workspace';
import { generateId } from '../utils/id';

export interface CreateEventResult {
  event: Event;
  workspace: Workspace;
}

export async function createEventWithWorkspace(
  input: CreateEventInput,
  ownerId: string
): Promise<CreateEventResult> {
  // Transactional: event + workspace created atomically.
  // If workspace creation fails, event is rolled back.
  return db.transaction(async (trx) => {
    const eventId = generateId('evt');
    const workspaceId = generateId('wsp');
    const now = new Date();

    const [event] = await trx('events')
      .insert({
        id: eventId,
        workspace_id: workspaceId,
        owner_id: ownerId,
        title: input.title,
        description: input.description ?? null,
        starts_at: new Date(input.startsAt),
        ends_at: new Date(input.endsAt),
        timezone: input.timezone,
        capacity: input.capacity ?? null,
        is_virtual: input.isVirtual,
        location: input.location ?? null,
        status: 'draft',
        created_at: now,
        updated_at: now,
      })
      .returning('*');

    const [workspace] = await trx('workspaces')
      .insert({
        id: workspaceId,
        event_id: eventId,
        owner_id: ownerId,
        name: `${input.title} Workspace`,
        created_at: now,
        updated_at: now,
      })
      .returning('*');

    return {
      event: mapEventRow(event),
      workspace: mapWorkspaceRow(workspace),
    };
  });
}

// ── Private mappers ──────────────────────────────────────────────────────────

function mapEventRow(row: Record<string, unknown>): Event {
  return {
    id: row.id as string,
    workspaceId: row.workspace_id as string,
    ownerId: row.owner_id as string,
    title: row.title as string,
    description: (row.description as string) ?? undefined,
    startsAt: new Date(row.starts_at as string),
    endsAt: new Date(row.ends_at as string),
    timezone: row.timezone as string,
    capacity: (row.capacity as number) ?? undefined,
    isVirtual: row.is_virtual as boolean,
    location: (row.location as string) ?? undefined,
    status: row.status as Event['status'],
    createdAt: new Date(row.created_at as string),
    updatedAt: new Date(row.updated_at as string),
  };
}

function mapWorkspaceRow(row: Record<string, unknown>): Workspace {
  return {
    id: row.id as string,
    eventId: row.event_id as string,
    ownerId: row.owner_id as string,
    name: row.name as string,
    createdAt: new Date(row.created_at as string),
    updatedAt: new Date(row.updated_at as string),
  };
}
```

### 3. Service Layer

**`src/services/event.service.ts`**
```typescript
import { CreateEventInput, CreateEventInputSchema } from '../models/event';
import { createEventWithWorkspace, CreateEventResult } from '../repositories/event.repository';
import { ValidationError, ConflictError } from '../errors';
import { logger } from '../utils/logger';

export async function createEvent(
  rawInput: unknown,
  ownerId: string
): Promise<CreateEventResult> {
  // 1. Validate input
  const parseResult = CreateEventInputSchema.safeParse(rawInput);
  if (!parseResult.success) {
    throw new ValidationError(
      'Invalid event input',
      parseResult.error.flatten().fieldErrors
    );
  }

  const input: CreateEventInput = parseResult.data;

  // 2. Business rule: event cannot start in the past (>5 min buffer for clock skew)
  const fiveMinutesAgo = new Date(Date.now() - 5 * 60 * 1000);
  if (new Date(input.startsAt) < fiveMinutesAgo) {
    throw new ValidationError('startsAt cannot be in the past', {
      startsAt: ['Event start time must be in the future'],
    });
  }

  // 3. Persist (atomic transaction in repository)
  logger.info({ ownerId, title: input.title }, 'Creating event');

  const result = await createEventWithWorkspace(input, ownerId);

  logger.info(
    { eventId: result.event.id, workspaceId: result.workspace.id },
    'Event created successfully'
  );

  return result;
}
```
