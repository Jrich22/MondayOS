## Store (Zustand)

```typescript
// src/features/event-workspace/store/createEventSlice.ts
import { create } from 'zustand';
import { immer } from 'zustand/middleware/immer';
import type {
  CreateEventFormState,
  CreateEventStep,
  EventDetails,
  WorkspaceConfig,
} from '../types/event.types';

interface CreateEventState {
  step: CreateEventStep;
  form: CreateEventFormState;
  isSubmitting: boolean;
  submissionError: string | null;

  // Actions
  setStep: (step: CreateEventStep) => void;
  updateDetails: (details: Partial<EventDetails>) => void;
  updateWorkspace: (workspace: Partial<WorkspaceConfig>) => void;
  setSubmitting: (value: boolean) => void;
  setSubmissionError: (error: string | null) => void;
  reset: () => void;
}

const INITIAL_STATE: Pick<
  CreateEventState,
  'step' | 'form' | 'isSubmitting' | 'submissionError'
> = {
  step: 'details',
  form: { details: {}, workspace: {} },
  isSubmitting: false,
  submissionError: null,
};

export const useCreateEventStore = create<CreateEventState>()(
  immer((set) => ({
    ...INITIAL_STATE,

    setStep: (step) => set((state) => { state.step = step; }),

    updateDetails: (details) =>
      set((state) => {
        state.form.details = { ...state.form.details, ...details };
        // Auto-derive workspace slug when name changes
        if (details.name !== undefined) {
          const derived = details.name
            .toLowerCase()
            .trim()
            .replace(/[^a-z0-9]+/g, '-')
            .replace(/^-|-$/g, '');
          if (!state.form.workspace.slug) {
            state.form.workspace.slug = derived;
          }
        }
      }),

    updateWorkspace: (workspace) =>
      set((state) => {
        state.form.workspace = { ...state.form.workspace, ...workspace };
      }),

    setSubmitting: (value) => set((state) => { state.isSubmitting = value; }),

    setSubmissionError: (error) =>
      set((state) => { state.submissionError = error; }),

    reset: () => set(() => ({ ...INITIAL_STATE })),
  }))
);
```

```typescript
// src/features/event-workspace/services/eventService.ts
import type { CreateEventPayload, EventRecord } from '../types/event.types';

// MVP_STUB: Replace with real HTTP client when API is wired.
// Simulates a ~300 ms network round-trip.
const MVP_OWNER_ID = 'dev-user-001';

function generateId(): string {
  return `evt_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
}

export async function createEvent(
  payload: CreateEventPayload
): Promise<EventRecord> {
  // Input guard — service should never receive invalid data
  if (!payload.details.name?.trim()) {
    throw new Error('Event name is required.');
  }
  if (!payload.workspace.slug?.trim()) {
    throw new Error('Workspace slug is required.');
  }

  // MVP: persist to localStorage so records survive page refresh
  await new Promise((r) => setTimeout(r, 300));

  const now = new Date().toISOString();
  const record: EventRecord = {
    id: generateId(),
    status: 'draft',
    createdAt: now,
    updatedAt: now,
    ownerId: MVP_OWNER_ID,
    details: { ...payload.details },
    workspace: { ...payload.workspace },
  };

  const existing = JSON.parse(
    localStorage.getItem('cue:events') ?? '[]'
  ) as EventRecord[];
  localStorage.setItem('cue:events', JSON.stringify([...existing, record]));

  return record;
}

export async function getEvents(): Promise<EventRecord[]> {
  await new Promise((r) => setTimeout(r, 100));
  return JSON.parse(localStorage.getItem('cue:events') ?? '[]');
}
```
