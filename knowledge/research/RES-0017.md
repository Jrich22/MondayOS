## 2. Validation (`src/lib/validation/createEventSchema.ts`)

```typescript
// src/lib/validation/createEventSchema.ts
import { z } from 'zod';

const IANA_TIMEZONES = Intl.supportedValuesOf('timeZone');

export const stepOneSchema = z.object({
  title: z
    .string()
    .min(3, 'Title must be at least 3 characters')
    .max(120, 'Title must be 120 characters or fewer'),
  description: z
    .string()
    .max(2000, 'Description must be 2 000 characters or fewer'),
});

export const stepTwoSchema = z
  .object({
    startDate: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'Invalid date'),
    startTime: z.string().regex(/^\d{2}:\d{2}$/, 'Invalid time'),
    endDate: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'Invalid date'),
    endTime: z.string().regex(/^\d{2}:\d{2}$/, 'Invalid time'),
    timezone: z.string().refine(
      (tz) => IANA_TIMEZONES.includes(tz),
      { message: 'Select a valid timezone' }
    ),
  })
  .refine(
    (data) => {
      const start = new Date(`${data.startDate}T${data.startTime}`);
      const end   = new Date(`${data.endDate}T${data.endTime}`);
      return end > start;
    },
    { message: 'End must be after start', path: ['endDate'] }
  );

export const stepThreeSchema = z.object({
  location: z.string().max(200, 'Location must be 200 characters or fewer'),
  capacity: z
    .string()
    .refine(
      (v) => v === '' || (/^\d+$/.test(v) && Number(v) >= 1),
      { message: 'Capacity must be a positive whole number or left blank' }
    ),
});

export const createEventSchema = stepOneSchema
  .merge(stepTwoSchema.innerType())   // unwrap the refine wrapper
  .merge(stepThreeSchema)
  .superRefine((data, ctx) => {
    // cross-schema: end-after-start (repeated here for full-form validation)
    const start = new Date(`${data.startDate}T${data.startTime}`);
    const end   = new Date(`${data.endDate}T${data.endTime}`);
    if (end <= start) {
      ctx.addIssue({
        code: z.ZodIssueCode.custom,
        message: 'End must be after start',
        path: ['endDate'],
      });
    }
  });
```

function readStore<T>(key: string): T[] {
  try {
    return JSON.parse(localStorage.getItem(key) ?? '[]') as T[];
  } catch {
    return [];
  }
}

function writeStore<T>(key: string, data: T[]): void {
  localStorage.setItem(key, JSON.stringify(data));
