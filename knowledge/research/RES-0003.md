  /** Unique key used as React key and aria labels */
  key: string;
  /** Human-readable label rendered in the nav */
  label: string;
  /** Path matched by React Router */
  path: string;
  /** Whether this link requires an exact path match for active state */
  exact?: boolean;
}

export interface AppRoute {
  path: string;
  element: React.ReactNode;
