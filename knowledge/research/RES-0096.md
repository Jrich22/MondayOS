## 1. Types (`src/features/checkin/types.ts`)

```typescript
// src/features/checkin/types.ts

export type AttendeeStatus = 'registered' | 'checked-in' | 'walk-in' | 'cancelled';

export type TicketTier = 'general' | 'vip' | 'speaker' | 'staff' | 'press';

export interface Attendee {
  id: string;
  qrCode: string;           // Unique token encoded in QR
  firstName: string;
  lastName: string;
  email: string;
  company?: string;
  title?: string;
  ticketTier: TicketTier;
  status: AttendeeStatus;
  checkedInAt?: string;     // ISO timestamp
  checkedInBy?: string;     // Staff member ID
  badgeNumber?: number;     // Assigned on check-in
  photoUrl?: string;        // Optional badge photo
  dietary?: string[];       // Badge/logistics data
  sessionAccess?: string[]; // Which sessions they can attend
  isWalkIn: boolean;
}

export interface CheckInEvent {
  id: string;
  name: string;
  date: string;             // ISO date
  venue: string;
  totalCapacity: number;
  registeredCount: number;
  checkedInCount: number;
}

export interface ScanResult {
  status: 'success' | 'duplicate' | 'not-found' | 'cancelled' | 'error';
  attendee?: Attendee;
  message: string;
  timestamp: string;
  scannedCode: string;
}

export interface CheckInAction {
  id: string;
  attendeeId: string;
  type: 'scan' | 'manual' | 'walk-in' | 'undo';
  result: ScanResult['status'];
  performedBy: string;
  timestamp: string;
}

export interface WalkInFormData {
  firstName: string;
  lastName: string;
  email: string;
  company?: string;
  ticketTier: TicketTier;
}

export interface BadgeData {
  attendeeId: string;
  firstName: string;
  lastName: string;
  company?: string;
  title?: string;
  ticketTier: TicketTier;
  badgeNumber: number;
  qrCode: string;
  eventName: string;
  sessionAccess?: string[];
}

export interface CheckInStats {
  total: number;
  checkedIn: number;
  walkIns: number;
  pending: number;
  checkInRate: number;      // 0–1
  recentScans: ScanResult[];
}

// Kiosk display state machine
export type KioskState =
  | 'idle'
  | 'scanning'
  | 'processing'
  | 'success'
  | 'duplicate'
  | 'not-found'
  | 'walk-in-form'
  | 'badge-preview';
```
