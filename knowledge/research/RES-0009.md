## Implementation Plan

### Design Decisions

1. **Interface-first**: Define `AuthService` interface before any implementation
2. **Fake implementation only**: `FakeAuthService` returns a hardcoded session — no JWT, no OAuth, no cookies
3. **Single injection point**: One place in the app wires the concrete implementation
4. **Testable by default**: The interface makes unit tests trivial to write
5. **Clearly marked**: Every fake file is annotated so no one ships this to production accidentally

```typescript
/**
 * AuthService interface — the seam.
 *
 * Any real auth implementation (Clerk, Auth0, Supabase Auth, custom JWT)
 * must satisfy this contract. The rest of the app depends ONLY on this
 * interface, never on a concrete class.
 *
 * MVP STATUS: Only FakeAuthService implements this today.
 * To swap in real auth: implement this interface, replace the binding
 * in auth.context.tsx. Nothing else changes.
 */

import { AuthSession, CurrentUser } from './auth.types';

export interface AuthService {
  /**
   * Returns the current session.
   * In the placeholder, always returns a hardcoded fake session.
   * In production, validates the token/cookie and returns the real user.
   */
  getSession(): Promise<AuthSession>;

  /**
   * Returns the current user, or null if no session exists.
   */
  getCurrentUser(): Promise<CurrentUser | null>;

  /**
   * Signals whether the service is operating with real identity.
   * Placeholder always returns false.
   * Useful for UI warnings during development.
   */
  isRealAuth(): boolean;
}
```

  session: AuthSession | null;
  isLoading: boolean;
  /** Exposes the service for advanced use-cases; prefer useAuth() hooks */
  service: AuthService;
}

const AuthContext = createContext<AuthContextValue | null>(null);

interface AuthProviderProps {
  children: ReactNode;
  /** Override the service — used in tests to inject a mock */
  serviceOverride?: AuthService;
}

export function AuthProvider({
  children,
  serviceOverride,
}: AuthProviderProps): JSX.Element {
  const service = serviceOverride ?? authService;

  const [session, setSession] = useState<AuthSession | null>(null);
  const [isLoading, setIsLoading] = useState<boolean>(true);

  useEffect(() => {
    let cancelled = false;

    service
      .getSession()
      .then((s) => {
        if (!cancelled) setSession(s);
      })
      .catch((err) => {
        // In production, log to error tracker; here just warn.
        console.warn('[AuthProvider] getSession failed:', err);
      })
      .finally(() => {
        if (!cancelled) setIsLoading(false);
      });

    return () => {
      cancelled = true;
    };
  }, [service]);

  return (
    <AuthContext.Provider value={{ session, isLoading, service }}>
      {children}
    </AuthContext.Provider>
  );
}
