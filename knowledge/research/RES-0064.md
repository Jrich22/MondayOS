# Cue App: Multi-Organization Architecture Design

## Executive Summary

This document defines the complete multi-organization architecture for the Cue App, enabling support for Venture Capital, Private Equity, Conferences, Corporate Events, Executive Offsites, Customer Advisory Boards, University Alumni, Nonprofits, and Associations. The design prioritizes extensibility, clean data modeling, and a clear path to multi-tenant production deployment.

---

## 1. Core Data Models

### 1.1 Organization Model

```typescript
// src/models/Organization.ts

export type OrganizationType =
  | 'venture_capital'
  | 'private_equity'
  | 'conference'
  | 'corporate_event'
  | 'executive_offsite'
  | 'customer_advisory_board'
  | 'university_alumni'
  | 'nonprofit'
  | 'association';

export type OrganizationStatus = 'active' | 'suspended' | 'onboarding' | 'churned';

export type BillingTier = 'free' | 'starter' | 'growth' | 'enterprise';

export interface OrganizationAddress {
  line1: string;
  line2?: string;
  city: string;
  state: string;
  postalCode: string;
  country: string; // ISO 3166-1 alpha-2
}

export interface OrganizationBranding {
  primaryColor: string;       // hex
  secondaryColor?: string;    // hex
  logoUrl?: string;
  faviconUrl?: string;
  customDomain?: string;      // e.g. "cue.acme.com"
  emailFromName?: string;
}

export interface Organization {
  // Identity
  id: string;                             // UUID v4
  slug: string;                           // URL-safe, globally unique: "sequoia-capital"
  name: string;
  type: OrganizationType;
  status: OrganizationStatus;

  // Profile
  description?: string;
  website?: string;
  address?: OrganizationAddress;
  foundedYear?: number;
  size?: OrganizationSizeRange;

  // Configuration
  industryProfileId: string;              // FK → IndustryProfile.id
  featureFlags: Record<string, boolean>;  // overrides on top of profile defaults
  customFields: CustomFieldValue[];       // org-level custom field values
  branding: OrganizationBranding;

  // Billing & Limits
  billingTier: BillingTier;
  billingEmail: string;
  maxWorkspaces: number;
  maxMembers: number;
  maxEventsPerMonth: number;

  // Multi-tenancy
  tenantId: string;                       // logical tenant grouping (1:1 initially)
  dataRegion: 'us-east-1' | 'eu-west-1' | 'ap-southeast-1';
  isolationLevel: 'shared' | 'dedicated_schema' | 'dedicated_db';

  // Audit
  createdAt: Date;
  updatedAt: Date;
  createdBy: string;                      // user ID
  deletedAt?: Date;                       // soft delete
}

export type OrganizationSizeRange =
  | '1-10'
  | '11-50'
  | '51-200'
  | '201-500'
  | '501-1000'
  | '1000+';
```

### 1.2 Workspace Model

A **Workspace** is the primary unit of collaboration within an Organization — an event, a fund cycle, a cohort, or a chapter.

```typescript
// src/models/Workspace.ts

export type WorkspaceType =
  // VC / PE
  | 'fund'
  | 'portfolio_review'
  | 'lp_meeting'
  // Events
  | 'conference'
  | 'summit'
  | 'workshop'
  | 'hackathon'
  // Corporate
  | 'offsite'
  | 'all_hands'
  | 'advisory_board_session'
  // Community
  | 'alumni_chapter'
  | 'cohort'
  | 'committee'
  // General
  | 'general';

export type WorkspaceVisibility = 'private' | 'internal' | 'public';
export type WorkspaceStatus = 'draft' | 'active' | 'archived' | 'cancelled';

export interface WorkspaceSchedule {
  timezone: string;           // IANA tz string, e.g. "America/New_York"
  startDate?: Date;
  endDate?: Date;
  isRecurring: boolean;
  recurrenceRule?: string;    // RFC 5545 RRULE
}

export interface Workspace {
  // Identity
  id: string;
  organizationId: string;     // FK → Organization.id
  slug: string;               // unique within org: "q4-2024-lp-meeting"
  name: string;
  type: WorkspaceType;
  status: WorkspaceStatus;
  visibility: WorkspaceVisibility;

  // Content
  description?: string;
  coverImageUrl?: string;
  tags: string[];

  // Schedule
  schedule?: WorkspaceSchedule;

  // Location
  venue?: WorkspaceVenue;

  // Configuration
  inheritedFeatureFlags: Record<string, boolean>;   // merged from org
  workspaceFeatureFlags: Record<string, boolean>;   // workspace-level overrides
  customFields: CustomFieldValue[];
  settings: WorkspaceSettings;

  // Capacity
  maxAttendees?: number;
  registrationEnabled: boolean;
  registrationDeadline?: Date;
  waitlistEnabled: boolean;

  // Ownership
  ownerId: string;            // user ID
  coOwnerIds: string[];

  // Audit
  createdAt: Date;
  updatedAt: Date;
  createdBy: string;
  deletedAt?: Date;
}

export interface WorkspaceVenue {
  name: string;
  address?: OrganizationAddress;
  virtualUrl?: string;
  isHybrid: boolean;
  mapEmbedUrl?: string;
}

export interface WorkspaceSettings {
  allowGuestRegistration: boolean;
  requireApproval: boolean;
  showAttendeeList: boolean;
  enableNetworking: boolean;
  enableAgenda: boolean;
  enableQA: boolean;
  enablePolling: boolean;
  enableSponsorSlots: boolean;
  notificationChannels: NotificationChannel[];
}

export type NotificationChannel = 'email' | 'sms' | 'push' | 'slack' | 'webhook';
```

## 2. Industry Profile System

An **IndustryProfile** is a reusable configuration template that ships with sensible defaults for each organization type. Organizations can override any field.

```typescript
// src/models/IndustryProfile.ts

export interface IndustryProfile {
  id: string;
  organizationType: OrganizationType;
  version: string;              // semver, e.g. "1.2.0"
  displayName: string;          // "Venture Capital"
  description: string;

  // What workspace types are valid for this org type
  allowedWorkspaceTypes: WorkspaceType[];
  defaultWorkspaceType: WorkspaceType;

  // Feature flag defaults for this profile
  defaultFeatureFlags: FeatureFlagSet;

  // Custom field schema to apply by default
  defaultCustomFieldSchemas: CustomFieldSchema[];

  // Role/permission presets
  defaultRolePresets: RolePreset[];

  // Onboarding flow steps
  onboardingSteps: OnboardingStep[];

  // UI hints
  terminology: TerminologyOverrides;

  // Navigation
  defaultNavItems: NavItemConfig[];

  createdAt: Date;
  updatedAt: Date;
}
