## 1. TypeScript Types

### `src/features/check-in/types/checkIn.types.ts`

```typescript
// ─── Attendee ────────────────────────────────────────────────────────────────

export type AttendeeStatus = 'registered' | 'checked-in' | 'walk-in' | 'cancelled';

export interface Attendee {
  id: string;                    // UUID — also encoded in QR
  firstName: string;
  lastName: string;
  email: string;
  company: string;
  jobTitle: string;
  ticketType: 'general' | 'vip' | 'speaker' | 'sponsor' | 'staff';
  status: AttendeeStatus;
  checkedInAt?: string;          // ISO 8601 timestamp
  checkedInBy?: string;          // Staff ID or 'kiosk'
  badgeColor: string;            // Hex — derived from ticketType
  dietaryRestrictions?: string;
  sessionAccess: string[];       // Session IDs attendee can enter
  photoUrl?: string;             // Optional headshot for badge
  isWalkIn: boolean;
}

// ─── QR Code ─────────────────────────────────────────────────────────────────

export interface QRPayload {
  attendeeId: string;
  eventId: string;
  ticketType: Attendee['ticketType'];
  checksum: string;              // Simple integrity check (not crypto-grade)
  version: number;               // QR schema version for future compat
}

export interface QRScanResult {
  raw: string;
  parsed: QRPayload | null;
  isValid: boolean;
  validationError?: string;
}

// ─── Check-In Flow ───────────────────────────────────────────────────────────

export type CheckInOutcome =
  | 'success'
  | 'duplicate'
  | 'not-found'
  | 'cancelled'
  | 'invalid-qr';

export interface CheckInAttempt {
  id: string;
  attendeeId: string | null;
  scannedQR: string;
  outcome: CheckInOutcome;
  timestamp: string;
  staffId: string;
}

export interface CheckInResult {
  outcome: CheckInOutcome;
  attendee: Attendee | null;
  attempt: CheckInAttempt;
  message: string;
}

// ─── Walk-In ──────────────────────────────────────────────────────────────────

export interface WalkInFormData {
  firstName: string;
  lastName: string;
  email: string;
  company: string;
  jobTitle: string;
  ticketType: Attendee['ticketType'];
}

export interface WalkInFormErrors {
  firstName?: string;
  lastName?: string;
  email?: string;
  company?: string;
  jobTitle?: string;
  ticketType?: string;
}

// ─── Badge ────────────────────────────────────────────────────────────────────

export interface BadgeData {
  attendeeId: string;
  displayName: string;
  company: string;
  jobTitle: string;
  ticketType: Attendee['ticketType'];
  badgeColor: string;
  qrCodeValue: string;
  sessionAccess: string[];
  photoUrl?: string;
}

// ─── Kiosk ────────────────────────────────────────────────────────────────────

export type KioskScreen =
  | 'idle'
  | 'scanning'
  | 'confirming'
  | 'success'
  | 'duplicate'
  | 'not-found'
  | 'walk-in'
  | 'badge-preview';

export interface KioskState {
  currentScreen: KioskScreen;
  idleTimeoutSeconds: number;
  lastActivityAt: string | null;
}

// ─── Store ────────────────────────────────────────────────────────────────────

export interface CheckInState {
  attendees: Record<string, Attendee>;
  scanHistory: CheckInAttempt[];
  currentScanResult: CheckInResult | null;
  kioskState: KioskState;
  isProcessing: boolean;
  error: string | null;
  stats: CheckInStats;
}

export interface CheckInStats {
  totalRegistered: number;
  totalCheckedIn: number;
  totalWalkIns: number;
  totalDuplicateScans: number;
}
```
