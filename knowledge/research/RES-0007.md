```typescript
/**
 * Core identity types for the Cue App auth seam.
 *
 * DESIGN NOTE: Keep this minimal. Real auth will implement the same
 * interface — nothing in the app should depend on the stub directly.
 */

export type UserId = string & { readonly _brand: "UserId" };

export interface CurrentUser {
  readonly id: UserId;
  readonly email: string;
  readonly displayName: string;
}

/**
 * The auth seam contract.
 *
 * Any real auth implementation (Clerk, Supabase Auth, NextAuth, etc.)
 * must satisfy this interface before the stub can be swapped out.
 */
export interface AuthProvider {
  /**
   * Returns the currently authenticated user.
   * Returns null if no session exists (unauthenticated).
   */
  getCurrentUser(): CurrentUser | null;

  /**
   * Returns true if there is an active session.
   */
  isAuthenticated(): boolean;
}
```

### `packages/cue-app/src/auth/current-user.ts`

```typescript
/**
 * The auth seam — single point of access to the current user.
 *
 * ALL application code must go through here. Never import StubAuthProvider
 * directly outside this module and the test suite.
 *
 * To swap in real auth:
 *   1. Implement AuthProvider with your real auth library
 *   2. Replace `activeProvider` below — nothing else changes
 *
 * This is the only file that needs to change at auth integration time.
 */

import { StubAuthProvider } from "./stub";
import type { AuthProvider, CurrentUser } from "./types";

// -------------------------------------------------------------------
// Seam — replace this line when wiring up real auth
// -------------------------------------------------------------------
let activeProvider: AuthProvider = new StubAuthProvider();
// -------------------------------------------------------------------

/**
 * Override the active provider. Intended for:
 *   - Dependency injection in tests
 *   - Future real-auth bootstrap at app startup
 *
 * @example
 *   // In tests:
 *   setAuthProvider(new StubUnauthenticatedProvider());
 *
 *   // In real auth bootstrap (future):
 *   setAuthProvider(new ClerkAuthProvider(clerkClient));
 */
export function setAuthProvider(provider: AuthProvider): void {
  activeProvider = provider;
}

/**
 * Returns the currently authenticated user, or null.
 * Use this everywhere you need identity in the app.
 */
export function getCurrentUser(): CurrentUser | null {
  return activeProvider.getCurrentUser();
}

/**
 * Returns true if a user session is active.
 */
export function isAuthenticated(): boolean {
  return activeProvider.isAuthenticated();
}

/**
 * Returns the current user, throwing if unauthenticated.
 * Use in contexts where auth is a hard precondition (e.g. API handlers
 * that sit behind an auth guard).
 *
 * @throws {Error} if no active session
 */
export function requireCurrentUser(): CurrentUser {
  const user = activeProvider.getCurrentUser();
  if (user === null) {
    throw new Error(
      "requireCurrentUser: no authenticated session. " +
        "Ensure this is called within an authenticated context."
    );
  }
  return user;
}
```
