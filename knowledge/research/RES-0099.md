## Data Models

### File: `src/types/checkin.ts`

```typescript
// ─────────────────────────────────────────────────────────────
// Cue App — Check-In Data Models (Sprint 2)
// All data is local/mock. No backend integration.
// ─────────────────────────────────────────────────────────────

export type AttendeeStatus =
  | 'registered'    // Pre-event, not yet checked in
  | 'checked_in'    // Successfully checked in
  | 'duplicate'     // Attempted second scan
  | 'walk_in'       // Added at door, no prior registration
  | 'cancelled';    // Registration cancelled

export type TicketTier =
  | 'general'
  | 'vip'
  | 'speaker'
  | 'sponsor'
  | 'staff'
  | 'press';

export interface Attendee {
  // Identity
  id: string;                        // UUID v4
  firstName: string;
  lastName: string;
  email: string;
  company?: string;
  jobTitle?: string;
  avatarInitials: string;            // Derived: "JD" for John Doe

  // Event
  eventId: string;
  ticketId: string;                  // Unique per ticket, used in QR
  ticketTier: TicketTier;
  registrationDate: string;          // ISO 8601

  // Check-in state
  status: AttendeeStatus;
  checkedInAt?: string;              // ISO 8601, set on check-in
  checkedInBy?: string;              // Staff ID or 'kiosk'
  checkInAttempts: number;           // Tracks duplicate scan count

  // Badge fields (badge-ready data model)
  badgeColor: string;                // Hex, derived from ticketTier
  badgeLabel: string;                // Human-readable tier label
  dietaryRestrictions?: string;
  accessibilityNeeds?: string;
  sessionInterests?: string[];

  // Walk-in specific
  isWalkIn: boolean;
  walkInCreatedAt?: string;          // ISO 8601
}

export interface QRPayload {
  ticketId: string;
  eventId: string;
  attendeeId: string;
  checksum: string;                  // Simple integrity check (Sprint 2: mock hash)
  version: '1';                      // Schema version for forward compatibility
}

export interface ScanResult {
  success: boolean;
  attendee?: Attendee;
  status: ScanStatus;
  message: string;
  timestamp: string;                 // ISO 8601
  scanDurationMs: number;            // For performance tracking (target: <300ms)
}

export type ScanStatus =
  | 'valid_new'           // First-time valid scan → check in
  | 'valid_duplicate'     // Already checked in → show warning
  | 'invalid_payload'     // QR data malformed
  | 'invalid_checksum'    // Checksum mismatch (tampered)
  | 'unknown_ticket'      // ticketId not in local store
  | 'wrong_event'         // eventId mismatch
  | 'cancelled_ticket'    // Attendee status is cancelled
  | 'walk_in_created';    // Walk-in successfully added

export interface WalkInFormData {
  firstName: string;
  lastName: string;
  email: string;
  company?: string;
  ticketTier: TicketTier;
}

export interface CheckInSession {
  id: string;
  eventId: string;
  operatorId: string;               // 'kiosk' or staff ID
  startedAt: string;                // ISO 8601
  totalScans: number;
  successfulCheckIns: number;
  duplicateScans: number;
  walkIns: number;
  lastActivity: string;             // ISO 8601
}

export interface BadgeData {
  attendeeId: string;
  fullName: string;
  company: string;
  jobTitle: string;
  ticketTier: TicketTier;
  badgeColor: string;
  badgeLabel: string;
  qrValue: string;                  // The encoded QR string
  eventName: string;
  eventDate: string;
}
```
