export const LIFECYCLE_STATES = [
  'planning',
  'inviting',
  'live',
  'follow_up',
  'insights',
] as const;

export type LifecycleState = typeof LIFECYCLE_STATES[number];

export const LIFECYCLE_TRANSITIONS = [
  'START_INVITING',
  'GO_LIVE',
  'END_EVENT',
  'COMPLETE_FOLLOWUP',
  'RESET_TO_PLANNING',
  'MANUAL_OVERRIDE',
] as const;

export type LifecycleTransition = typeof LIFECYCLE_TRANSITIONS[number];

export interface LifecycleContext {
  eventId: string;
  eventName: string;
  scheduledStart: Date | null;
  scheduledEnd: Date | null;
  attendeeCount: number;
  rsvpCount: number;
  followUpDeadline: Date | null;
  createdAt: Date;
  lastTransitionAt: Date;
  manualOverride: boolean;
  overriddenBy: string | null;
}

export interface LifecycleTransitionRecord {
  id: string;
  fromState: LifecycleState;
  toState: LifecycleState;
  transition: LifecycleTransition;
  triggeredBy: 'auto' | 'manual';
  triggeredByUserId?: string;
  timestamp: Date;
  reason?: string;
}

export interface WorkspaceConfig {
  state: LifecycleState;
  primaryColor: string;
  headerBadge: string;
  headerBadgeVariant: 'default' | 'success' | 'warning' | 'destructive' | 'live';
  enabledNavItems: NavItem[];
  pinnedWidgets: WidgetConfig[];
  availableWidgets: WidgetConfig[];
  aiActions: AIAction[];
  dashboardLayout: DashboardLayout;
  bannerConfig: BannerConfig | null;
}

export interface NavItem {
  id: string;
  label: string;
  icon: string;
  route: string;
  badge?: string | number;
  highlighted?: boolean;
}

export interface WidgetConfig {
  id: string;
  type: string;
  title: string;
  size: 'sm' | 'md' | 'lg' | 'xl';
  position: { row: number; col: number };
  props?: Record<string, unknown>;
  priority: number;
}

export interface AIAction {
  id: string;
  label: string;
  description: string;
  icon: string;
  prompt: string;
  category: 'draft' | 'analyze' | 'suggest' | 'summarize' | 'engage';
  isPrimary: boolean;
}

export interface DashboardLayout {
  id: string;
  name: string;
  columns: number;
  sections: DashboardSection[];
}

export interface DashboardSection {
  id: string;

  widgets: string[]; // widget IDs

  defaultCollapsed: boolean;
}

export interface BannerConfig {
