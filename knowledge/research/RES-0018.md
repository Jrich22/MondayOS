### 2.1 Zod Schema (`createEvent.schema.ts`)

```typescript
// src/features/create-event/schema/createEvent.schema.ts
import { z } from "zod";

export const eventDetailsSchema = z.object({
  name: z
    .string()
    .min(2, "Event name must be at least 2 characters")
    .max(120, "Event name must be 120 characters or fewer")
    .trim(),
  dateStart: z
    .string()
    .datetime({ message: "Start date must be a valid ISO datetime" }),
  dateEnd: z
    .string()
    .datetime({ message: "End date must be a valid ISO datetime" })
    .optional(),
  venue: z
    .string()
    .max(200, "Venue must be 200 characters or fewer")
    .trim()
    .optional(),
}).refine(
  (data) => {
    if (!data.dateEnd) return true;
    return new Date(data.dateEnd) > new Date(data.dateStart);
  },
  { message: "End date must be after start date", path: ["dateEnd"] }
);

export const workspaceSetupSchema = z.object({
  slug: z
    .string()
    .min(3, "Slug must be at least 3 characters")
    .max(48, "Slug must be 48 characters or fewer")
    .regex(/^[a-z0-9-]+$/, "Slug may only contain lowercase letters, numbers, and hyphens")
    .trim(),
  workspaceType: z.enum(["conference", "festival", "corporate", "community"], {
    errorMap: () => ({ message: "Select a valid workspace type" }),
  }),
});

export const createEventInputSchema = eventDetailsSchema.merge(workspaceSetupSchema);

export type EventDetailsInput = z.infer<typeof eventDetailsSchema>;
export type WorkspaceSetupInput = z.infer<typeof workspaceSetupSchema>;
export type CreateEventInput = z.infer<typeof createEventInputSchema>;
```

### 3.1 Workspace Factory (`provisionWorkspace.ts`)

```typescript
// src/lib/workspace/provisionWorkspace.ts
import type { WorkspaceSetupInput } from "@/features/create-event/schema/createEvent.schema";

export interface WorkspaceProvisionInput extends WorkspaceSetupInput {
  eventId: string;
  ownerId: string;
}

export interface ProvisionedWorkspace {
  id: string;
  slug: string;
  workspaceType: string;
  eventId: string;
  ownerId: string;
  createdAt: Date;
}

/**
 * Pure factory — produces a workspace record shape.
 * Persistence is the caller's responsibility (tRPC router).
 * Separated so it can be unit-tested without a DB connection.
 */
export function buildWorkspaceRecord(
  input: WorkspaceProvisionInput,
  idGenerator: () => string = () => crypto.randomUUID()
): ProvisionedWorkspace {
  return {
    id: idGenerator(),
    slug: input.slug.toLowerCase(),
    workspaceType: input.workspaceType,
    eventId: input.eventId,
    ownerId: input.ownerId,
    createdAt: new Date(),
  };
}
```
