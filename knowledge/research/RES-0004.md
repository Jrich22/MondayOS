### `src/test/renderWithRouter.tsx`
```typescript
import { render, type RenderResult } from '@testing-library/react';
import { MemoryRouter, type MemoryRouterProps } from 'react-router-dom';
import type { ReactNode } from 'react';

interface Options extends MemoryRouterProps {
  children?: ReactNode;
}

/**
 * Wraps a component in MemoryRouter so route-dependent components
 * render correctly in unit tests.
 */
export function renderWithRouter(
  ui: ReactNode,
  { initialEntries = ['/'], ...rest }: Options = {}
): RenderResult {
  return render(
    <MemoryRouter initialEntries={initialEntries} {...rest}>
      {ui}
    </MemoryRouter>
  );
}
```
