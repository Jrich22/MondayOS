/**
 * ReqWorkspace — the central organizing unit for a recruiting engagement.
 * One workspace per open requisition.
 *
 * State machine:
 *   draft → active → paused → closed | archived
 */

const REQ_STATES = Object.freeze({
  DRAFT:    'draft',
  ACTIVE:   'active',
  PAUSED:   'paused',
  CLOSED:   'closed',
  ARCHIVED: 'archived',
});

const VALID_TRANSITIONS = Object.freeze({
  [REQ_STATES.DRAFT]:    [REQ_STATES.ACTIVE],
  [REQ_STATES.ACTIVE]:   [REQ_STATES.PAUSED, REQ_STATES.CLOSED],
  [REQ_STATES.PAUSED]:   [REQ_STATES.ACTIVE, REQ_STATES.CLOSED],
  [REQ_STATES.CLOSED]:   [REQ_STATES.ARCHIVED],
  [REQ_STATES.ARCHIVED]: [],
});

class ReqWorkspace {
  constructor({
    id,
    title,
    hiringManager,

    targetCloseDate,
    state = REQ_STATES.DRAFT,
    linkedBriefIds = [],
    sessionIds = [],
    createdAt = new Date().toISOString(),
    updatedAt = new Date().toISOString(),
  }) {
    this.id = id;
    this.title = title;
    this.hiringManager = hiringManager;
    this.department = department;
    this.location = location;

    this.targetHeadcount = targetHeadcount;
    this.openDate = openDate;
    this.targetCloseDate = targetCloseDate;
    this.state = state;
    this.linkedBriefIds = linkedBriefIds;

    this.createdAt = createdAt;
    this.updatedAt = updatedAt;
