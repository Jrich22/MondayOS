# Cue App: Mission Control Dashboard

## Engineering Implementation Plan & Deliverables

---

## Executive Summary

This document delivers the complete engineering specification, component architecture, data contracts, and implementation artifacts for the **Mission Control Dashboard** — Cue App's organizer home screen. The design prioritizes immediate utility for single-event operation while building the structural foundation for organization-wide and portfolio-wide expansion.

---

## 1. Architecture Overview

### 1.1 Design Philosophy

The dashboard is built on three architectural principles:

1. **Scope Progression**: Single Event → Organization → Portfolio views via a unified data abstraction layer
2. **Real-time First**: WebSocket-driven live updates with optimistic UI and graceful degradation to polling
3. **AI-Augmented Decisioning**: Recommendations surface inline, contextual to current scope and health state

## 2. Data Contracts

### 2.1 Core Domain Types

```typescript
// src/types/mission-control.ts

export type HealthStatus = 'healthy' | 'warning' | 'critical' | 'unknown';

export interface EventHealth {
  eventId: string;
  eventName: string;
  status: HealthStatus;
  score: number;           // 0–100
  checkedAt: string;       // ISO-8601
  dimensions: {
    attendance: HealthDimension;
    revenue:    HealthDimension;
    logistics:  HealthDimension;
    engagement: HealthDimension;
    staffing:   HealthDimension;
  };
}

export interface HealthDimension {
  status: HealthStatus;
  score:  number;
  delta:  number;          // change vs prior period
  label:  string;
  detail: string;
}

export interface AttendanceSnapshot {
  eventId:      string;
  capacity:     number;
  registered:   number;
  checkedIn:    number;
  waitlisted:   number;
  cancelledToday: number;
  velocityPerHour: number;  // check-ins/hour rolling 60m
  occupancyPct: number;
  trend:        'accelerating' | 'steady' | 'decelerating';
  lastUpdated:  string;
}

export interface Alert {
  id:         string;
  eventId:    string;
  severity:   'info' | 'warning' | 'critical';
  category:   'attendance' | 'revenue' | 'logistics' | 'staff' | 'system';
  title:      string;
  body:       string;
  action?:    AlertAction;
  createdAt:  string;
  resolvedAt: string | null;
  snoozedUntil: string | null;
}

export interface AlertAction {
  label:    string;
  href?:    string;
  handler?: string;   // named handler key for in-app actions
}

export interface AIRecommendation {
  id:         string;
  scope:      DashboardScope;
  priority:   'high' | 'medium' | 'low';
  category:   string;
  headline:   string;
  rationale:  string;
  impact:     string;
  confidence: number;   // 0–1
  actions:    RecommendationAction[];
  generatedAt: string;
  expiresAt:   string;
  dismissed:   boolean;
}

export interface RecommendationAction {
  label:     string;
  variant:   'primary' | 'secondary' | 'destructive';
  handler:   string;
  payload?:  Record<string, unknown>;
}

export interface Milestone {
  id:          string;
  eventId:     string;
  title:       string;
  dueAt:       string;
  completedAt: string | null;
  status:      'on-track' | 'at-risk' | 'overdue' | 'complete';
  owner:       string;
  category:    'venue' | 'marketing' | 'logistics' | 'finance' | 'content' | 'other';
  daysUntilDue: number;
}

export interface OrgMetrics {
  orgId:            string;
  period:           'day' | 'week' | 'month' | 'quarter';
  activeEvents:     number;
  totalAttendees:   number;
  grossRevenue:     number;
  netRevenue:       number;
  avgHealthScore:   number;
  eventsAtRisk:     number;
  npsScore:         number | null;
  revenueVsBudget:  number;   // ratio, e.g. 1.05 = 5% over
  topPerformingEvent: string;
}

export interface DashboardState {
  scope:           DashboardScope;
  health:          EventHealth | EventHealth[];   // array for org/portfolio scope
  attendance:      AttendanceSnapshot | AttendanceSnapshot[];
  alerts:          Alert[];
  recommendations: AIRecommendation[];
  milestones:      Milestone[];
  orgMetrics?:     OrgMetrics;
  lastRefreshed:   string;
  connectionStatus: 'live' | 'polling' | 'stale' | 'offline';
}
```
