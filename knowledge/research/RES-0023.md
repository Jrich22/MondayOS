### 3.3 `useEvent.ts`

```typescript
import { useState, useEffect } from 'react';
import { Event } from '../types/event.types';
import { getEventById } from '../fixtures/events.fixture';

export type UseEventState =
  | { status: 'loading' }
  | { status: 'found'; event: Event }
  | { status: 'not_found' }
  | { status: 'error'; message: string };

/**
 * Resolves a single event by ID.
 *
 * MVP implementation uses local fixture data.
 * The interface is designed to be a drop-in replacement for a real
 * API call — callers depend on the state shape, not the data source.
 *
 * @param eventId - The route param from /events/:eventId
 */
export function useEvent(eventId: string | undefined): UseEventState {
  const [state, setState] = useState<UseEventState>({ status: 'loading' });

  useEffect(() => {
    if (!eventId) {
      setState({ status: 'not_found' });
      return;
    }

    // Simulate async resolution so the loading state is exercisable in tests
    const timer = setTimeout(() => {
      try {
        const event = getEventById(eventId);
        setState(event ? { status: 'found', event } : { status: 'not_found' });
      } catch (err) {
        setState({ status: 'error', message: 'Failed to load event.' });
      }
    }, 0);

    return () => clearTimeout(timer);
  }, [eventId]);

  return state;
}
```
