# Build the App Shell: Navigation + Layout

## Cue App — Sprint 1 MVP Deliverable

---

## 1. Architecture Decision

Before writing code, I'm making the following scoped decisions consistent with the MVP constraints (no payments, no integrations, no production auth):

- **Router**: React Router v6 (declarative, nested routes)
- **Layout**: Single shell component with persistent top nav + sidebar
- **Auth**: Placeholder/mock auth context only (no real auth provider)
- **Styling**: CSS Modules + CSS custom properties (no external UI lib dependency to keep the shell lean)
- **State**: React Context for nav state; no Redux at this stage

### 3.3 `src/shell/ShellContext.tsx` — Navigation State

```tsx
// src/shell/ShellContext.tsx
// Manages sidebar collapsed state and active nav item.
// Kept in Context (not global store) — shell-local concern.

import React, {
  createContext,
  useCallback,
  useContext,
  useMemo,
  useState,
} from 'react';

interface ShellContextValue {
  isSideNavCollapsed: boolean;
  toggleSideNav: () => void;
  collapseSideNav: () => void;
}

const ShellContext = createContext<ShellContextValue | null>(null);

export const ShellProvider: React.FC<{ children: React.ReactNode }> = ({
  children,
}) => {
  const [isSideNavCollapsed, setIsSideNavCollapsed] = useState(false);

  const toggleSideNav = useCallback(
    () => setIsSideNavCollapsed((prev) => !prev),
    []
  );

  const collapseSideNav = useCallback(() => setIsSideNavCollapsed(true), []);

  const value = useMemo(
    () => ({ isSideNavCollapsed, toggleSideNav, collapseSideNav }),
    [isSideNavCollapsed, toggleSideNav, collapseSideNav]
  );

  return (
    <ShellContext.Provider value={value}>{children}</ShellContext.Provider>
  );
};

export const useShell = (): ShellContextValue => {
  const ctx = useContext(ShellContext);
  if (!ctx) {
    throw new Error('useShell must be used within ShellProvider');
  }
  return ctx;
};
```

### 3.4 `src/shell/TopNav/TopNav.tsx`

```tsx
// src/shell/TopNav/TopNav.tsx
// Persistent top navigation bar.
// Renders: hamburger toggle, app logo/wordmark, user avatar (from MockAuth).

import React from 'react';
import { Link } from 'react-router-dom';
import { useMockAuth } from '../../shared/auth/MockAuthContext';
import { useShell } from '../ShellContext';
import { ROUTES } from '../../app/routes';
import styles from './TopNav.module.css';

const TopNav: React.FC = () => {
  const { user } = useMockAuth();
  const { toggleSideNav, isSideNavCollapsed } = useShell();

  return (
    <header className={styles.topNav} role="banner">
      <div className={styles.left}>
        <button
          className={styles.hamburger}
          onClick={toggleSideNav}
          aria-label={isSideNavCollapsed ? 'Expand sidebar' : 'Collapse sidebar'}
          aria-expanded={!isSideNavCollapsed}
          aria-controls="side-nav"
        >
          <span className={styles.hamburgerBar} />
          <span className={styles.hamburgerBar} />
          <span className={styles.hamburgerBar} />
        </button>

        <Link
          to={ROUTES.EVENTS}
          className={styles.logo}
          aria-label="Cue — go to events dashboard"
        >
          <span className={styles.logoMark} aria-hidden="true">◆</span>
          <span className={styles.logoWord}>Cue</span>
        </Link>
      </div>

      <nav className={styles.right} aria-label="User menu">
        <div
          className={styles.avatar}
          aria-label={`Signed in as ${user.name}`}
          title={user.email}
        >
          {user.avatarInitials}
        </div>
      </nav>
    </header>
  );
};

export default TopNav;
```
