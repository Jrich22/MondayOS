/**
 * Centralised route definitions for Cue App.
 * Single source of truth — avoids magic strings across components.

export const ROUTES = {
  ROOT: '/',
  EVENTS: {
    LIST: '/events',
    DETAIL: (id: string) => `/events/${id}`,

### 3.5 AppShell Component — `src/components/shell/AppShell.tsx`

```tsx
import type { ReactNode } from 'react';
import { TopNav } from './TopNav';
import { SideNav } from './SideNav';

interface AppShellProps {
  children: ReactNode;
}

/**
 * AppShell composes the full navigation frame.
 *
 * Layout:
 * ┌─────────────────────────────────────────┐
 * │              TopNav (64px)              │
 * ├──────────┬──────────────────────────────┤
 * │ SideNav  │        <main>                │
 * │ (240px)  │    page content here         │
 * │          │                              │
 * └──────────┴──────────────────────────────┘
 */
export function AppShell({ children }: AppShellProps) {
  return (
    <div className="shell-root" data-testid="app-shell">
      <TopNav />
      <div className="shell-body">
        <SideNav />
        <main className="shell-main" id="main-content" tabIndex={-1}>
          {children}
        </main>
      </div>
    </div>
  );
}
```
