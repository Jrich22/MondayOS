// src/types/event.ts

export type EventStatus =
  | 'draft'
  | 'published'
  | 'on_sale'
  | 'sold_out'
  | 'cancelled'
  | 'completed';

export interface CueEvent {
  id: string;
  title: string;
  slug: string;
  status: EventStatus;
  venueId: string;
  venueName: string;
  startsAt: string;          // ISO 8601
  endsAt: string;            // ISO 8601
  timezone: string;          // IANA tz, e.g. "Europe/London"
  ticketsSold: number;
  capacity: number;
  createdAt: string;
  updatedAt: string;
}

// Derived helpers — keeps UI logic out of components
export function isSoldOut(event: CueEvent): boolean {
  return event.ticketsSold >= event.capacity;
}

export function occupancyPercent(event: CueEvent): number {
  if (event.capacity === 0) return 0;
  return Math.round((event.ticketsSold / event.capacity) * 100);
}

export const STATUS_LABELS: Record<EventStatus, string> = {
  draft: 'Draft',
  published: 'Published',
  on_sale: 'On Sale',
  sold_out: 'Sold Out',
  cancelled: 'Cancelled',
  completed: 'Completed',
};
