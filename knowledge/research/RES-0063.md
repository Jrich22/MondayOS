# Cue App: Multi-Organization Architecture Design

## Executive Summary

This document defines the complete multi-organization architecture for the Cue App, enabling support for diverse organization types including Venture Capital, Private Equity, Conferences, Corporate Events, Executive Offsites, Customer Advisory Boards, University Alumni, Nonprofits, and Associations. This is a **design-only deliverable** — no production backend or auth implementation is included.

---

## 1. Organization Model

### 1.1 Core Organization Entity

```typescript
// types/organization.ts

export type OrganizationType =
  | 'venture_capital'
  | 'private_equity'
  | 'conference'
  | 'corporate_event'
  | 'executive_offsite'
  | 'customer_advisory_board'
  | 'university_alumni'
  | 'nonprofit'
  | 'association';

export type OrganizationStatus = 'active' | 'suspended' | 'trial' | 'archived';

export type BillingTier = 'free' | 'starter' | 'growth' | 'enterprise';

export interface Organization {
  // Identity
  id: string;                          // UUID v4
  slug: string;                        // URL-safe unique identifier, e.g. "sequoia-cap"
  displayName: string;
  legalName?: string;
  logoUrl?: string;
  websiteUrl?: string;

  // Classification
  type: OrganizationType;
  industryProfileId: string;           // FK → IndustryProfile
  subtype?: string;                    // e.g. "seed_stage" for VC, "buyout" for PE

  // Status & Billing
  status: OrganizationStatus;
  billingTier: BillingTier;
  trialExpiresAt?: Date;

  // Feature & Permission Flags
  featureFlags: FeatureFlagSet;        // see §4
  permissionPolicy: PermissionPolicy;  // see §5

  // Custom Configuration
  configuredFields: ConfiguredField[]; // see §3
  workspaceIds: string[];              // FK[] → Workspace

  // Multi-Tenant Metadata
  tenantId: string;                    // logical tenant boundary (see §7)
  dataRegion: DataRegion;

  // Audit
  createdAt: Date;
  updatedAt: Date;
  createdBy: string;                   // user ID
  ownerId: string;                     // user ID, primary admin
}

export type DataRegion = 'us-east-1' | 'eu-west-1' | 'ap-southeast-1';
```
