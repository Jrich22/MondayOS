## 2. Type Definitions

**`src/types/invitation.ts`**

```typescript
/**
 * Invitation entity — owns lifecycle/token for a single guest.
 * Guest entity remains canonical for RSVP status, actual plus-ones,
 * attendance, and preferences. No Event or Guest fields duplicated here.
 * Per DEC-0010 / DEC-0011 / DOC-0016.
 */

export type InvitationStatus =
  | 'draft'      // issued but not yet delivered
  | 'delivered'  // simulated delivery sent
  | 'responded'  // guest has submitted a response
  | 'revoked'    // organizer revoked — token invalid
  | 'rotated';   // organizer rotated — prior token invalid, new active

export type RSVPChoice = 'confirmed' | 'tentative' | 'declined';

/** Immutable token snapshot kept for audit. */
export interface InvitationToken {
  readonly id: string;          // uuid — the URL-safe token
  readonly version: number;     // increments on each rotate
  readonly issuedAt: string;    // ISO-8601
  readonly invalidatedAt?: string; // set on rotate/revoke
}

export interface Invitation {
  readonly id: string;              // invitation uuid
  readonly eventId: string;
  readonly guestId: string;

  // ── Token (current active token is tokens[tokens.length - 1]) ──
  tokens: InvitationToken[];        // full history; only latest is active

  // ── Lifecycle ──
  status: InvitationStatus;
  createdAt: string;                // ISO-8601
  revokedAt?: string;
  rotatedAt?: string;
  respondedAt?: string;
  deliveredAt?: string;             // simulated

  // ── Plus-one allowance (per-invitation, default 0) ──
  plusOneAllowance: number;         // non-negative integer

  // ── Simulated delivery ──
  deliverySimulated: boolean;
  deliveryChannel: 'link' | 'email-sim' | 'sms-sim';
}

/** What the guest submits — applied to canonical Guest entity. */
export interface RSVPSubmission {
  invitationId: string;
  tokenId: string;                  // must match current active token
  choice: RSVPChoice;
  plusOneCount: number;             // 0 … plusOneAllowance
  dietaryNotes: string;             // synthetic / free-text
  accessibilityNotes: string;       // synthetic / free-text
}

/** Reasons the guest response surface can be blocked. */
export type GuestSurfaceBlockReason =
  | 'revoked'
  | 'rotated'           // token stale — a newer token exists
  | 'invalid-token'
  | 'event-started'
  | 'capacity-blocked'  // no waitlist and event is full
  | 'waitlisted';       // accepted but on waitlist (no auto-promotion)
```

      if (event && new Date(event.startAt) <= new Date()) {
        return { invitation: inv, blockReason: 'event-started' };
      }

      return { invitation: inv, blockReason: null };
    }

    return { invitation: null, blockReason: 'invalid-token' };
