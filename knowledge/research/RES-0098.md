# Cue App: QR & Badge Check-In — Sprint 2 Implementation

## Executive Summary

This document delivers the complete Sprint 2 implementation for QR & Badge Check-In in the Cue App. All features are local/mock only — no camera hardware, printer integration, backend, or production auth. The implementation covers the full attendee lifecycle: QR generation → scan simulation → validation → check-in → badge preview → kiosk mode.

## 1. Domain Types

**`src/features/check-in/types/index.ts`**

```typescript
// ─── Attendee ────────────────────────────────────────────────────────────────

export type TicketTier = 'general' | 'vip' | 'speaker' | 'staff' | 'press';

export interface Attendee {
  id: string;                        // UUID v4
  eventId: string;
  firstName: string;
  lastName: string;
  email: string;
  company: string | null;
  title: string | null;
  ticketTier: TicketTier;
  ticketNumber: string;              // Human-readable, e.g. "TKT-0042"
  qrPayload: string;                 // Signed opaque token
  avatarInitials: string;            // Derived: "JD"
  dietaryNotes: string | null;
  accessibilityNotes: string | null;
  registeredAt: string;              // ISO-8601
}

// ─── Check-In State ──────────────────────────────────────────────────────────

export type CheckInStatus =
  | 'not_checked_in'
  | 'checked_in'
  | 'duplicate'
  | 'invalid';

export interface CheckInRecord {
  attendeeId: string;
  checkedInAt: string;               // ISO-8601
  method: 'qr_scan' | 'badge_tap' | 'manual' | 'walk_in';
  operatorId: string;
  deviceId: string;
}

export interface AttendeeCheckInState {
  attendee: Attendee;
  status: CheckInStatus;
  record: CheckInRecord | null;
}

// ─── Scan ────────────────────────────────────────────────────────────────────

export type ScanOutcome =
  | { kind: 'success';   attendee: Attendee; alreadyCheckedIn: false }
  | { kind: 'duplicate'; attendee: Attendee; alreadyCheckedIn: true;  record: CheckInRecord }
  | { kind: 'invalid';   reason: string }
  | { kind: 'not_found'; rawPayload: string };

// ─── Walk-In ─────────────────────────────────────────────────────────────────

export interface WalkInFormValues {
  firstName: string;
  lastName: string;
  email: string;
  company: string;
  ticketTier: TicketTier;
}

// ─── Badge ───────────────────────────────────────────────────────────────────

export interface BadgeData {
  attendeeId: string;
  fullName: string;
  company: string;
  title: string;
  ticketTier: TicketTier;
  ticketNumber: string;
  qrPayload: string;
  avatarInitials: string;
  checkedInAt: string;
  eventName: string;
  eventDate: string;
  tierColor: string;                 // CSS hex
  tierLabel: string;
}

// ─── Store ───────────────────────────────────────────────────────────────────

export interface CheckInStats {
  total: number;
  checkedIn: number;
  walkIns: number;
  byTier: Record<TicketTier, { total: number; checkedIn: number }>;
}

export interface CheckInStore {
  attendees: AttendeeCheckInState[];
  recentScans: ScanOutcome[];
  stats: CheckInStats;
  isKioskMode: boolean;
  activeEventId: string;

  // Actions
  processQRScan: (rawPayload: string) => ScanOutcome;
  checkInAttendee: (attendeeId: string, method: CheckInRecord['method']) => CheckInRecord;
  addWalkIn: (values: WalkInFormValues) => Attendee;
  resetAttendeeCheckIn: (attendeeId: string) => void;
  setKioskMode: (enabled: boolean) => void;
  clearRecentScans: () => void;
}
```
