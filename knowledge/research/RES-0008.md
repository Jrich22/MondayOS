## Design Decisions (Brief)

1. **Interface-first**: `AuthService` interface defined first; fake impl satisfies it
2. **Single source of truth**: One `getCurrentUser()` call site, no scattered constants
3. **Zero real auth surface**: No JWT, no cookies, no network calls — purely in-memory
4. **Drop-in ready**: Real auth replaces only the provider binding, not call sites
5. **Typed, tested, documented**

import type { AuthService } from './auth.interface';
import type { Session, User } from './auth.types';

/**
 * FakeAuthService — MVP placeholder implementation of AuthService.
 *
 * ⚠️  NOT FOR PRODUCTION USE.
 * - No real credentials are checked.
 * - No tokens are issued or validated.
 * - Session state is held in memory only (lost on reload).
 *
 * Purpose: unblock UI development and test wiring before real auth exists.
 * Replacement path: swap provider binding in auth.provider.ts only.
 */

const FAKE_USER: User = Object.freeze({
  id: 'fake-user-001',
  displayName: 'Dev User',
  email: 'dev@cue.local',
  roles: ['admin'] as const,
});

const FAKE_SESSION: Session = Object.freeze({
  user: FAKE_USER,
  // Far-future expiry so MVP flows never hit expiry edge cases
  expiresAt: '2099-12-31T23:59:59.000Z',
  isAuthenticated: false, // Explicitly false — this is NOT real auth
});

export class FakeAuthService implements AuthService {
  private _session: Session | null = FAKE_SESSION;

  getSession(): Session | null {
    return this._session;
  }

  getCurrentUser(): User | null {
    return this._session?.user ?? null;
  }

  isAuthenticated(): boolean {
    // Placeholder always returns true so UI flows work,
    // but session.isAuthenticated remains false for honesty.
    return this._session !== null;
  }

  /**
   * Fake sign-in: accepts any credentials, returns fake session.
   * Logs a clear warning so developers know this isn't real.
   */
  async signIn(_email: string, _password: string): Promise<Session> {
    if (process.env.NODE_ENV !== 'test') {
      console.warn(
        '[FakeAuthService] signIn() called — no credentials validated. ' +
          'Replace FakeAuthService before production.',
      );
    }
    this._session = FAKE_SESSION;
    return FAKE_SESSION;
  }

  async signOut(): Promise<void> {
    this._session = null;
  }
}
