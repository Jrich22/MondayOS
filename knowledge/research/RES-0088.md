## Core Types

```typescript
// src/features/communications/types.ts

export type CommunicationType =
  | 'invitation'
  | 'rsvp_reminder'
  | 'event_update'
  | 'thank_you'
  | 'post_event_survey';

export type CommunicationStatus =
  | 'draft'
  | 'scheduled'
  | 'sending'
  | 'sent'
  | 'failed'
  | 'cancelled';

export type AudienceSegment =
  | 'all_guests'
  | 'confirmed'
  | 'declined'
  | 'pending'
  | 'vip'
  | 'speakers'
  | 'sponsors'
  | 'waitlist'
  | 'custom';

export interface Recipient {
  id: string;
  name: string;
  email: string;
  rsvpStatus: 'confirmed' | 'declined' | 'pending' | 'waitlist';
  tags: string[];
  segment: AudienceSegment[];
}

export interface AudienceFilter {
  segments: AudienceSegment[];
  tags?: string[];
  excludeIds?: string[];
  customRule?: string;
}

export interface CommunicationTemplate {
  id: string;
  name: string;
  type: CommunicationType;
  subject: string;
  bodyHtml: string;
  bodyText: string;
  variables: TemplateVariable[];
  tags: string[];
  usageCount: number;
  createdAt: string;
  updatedAt: string;
  isSystem: boolean;
}

export interface TemplateVariable {
  key: string;           // e.g. "{{guest_name}}"
  label: string;         // e.g. "Guest Name"
  defaultValue: string;
  required: boolean;
}

export interface ScheduleConfig {
  sendAt: string | null;       // ISO datetime, null = send now
  timezone: string;
  recurrence: RecurrenceRule | null;
  sendCondition: SendCondition | null;
}

export interface RecurrenceRule {
  type: 'daily' | 'weekly' | 'custom';
  intervalDays?: number;
  daysOfWeek?: number[];
  endAt?: string;
  maxOccurrences?: number;
}

export interface SendCondition {
  type: 'days_before_event' | 'days_after_event' | 'on_rsvp' | 'manual';
  offsetDays?: number;
}

export interface Campaign {
  id: string;
  eventId: string;
  name: string;
  type: CommunicationType;
  status: CommunicationStatus;
  subject: string;
  bodyHtml: string;
  bodyText: string;
  audience: AudienceFilter;
  resolvedRecipients: Recipient[];
  schedule: ScheduleConfig;
  templateId: string | null;
  aiGenerated: boolean;
  aiPrompt?: string;
  sentCount: number;
  openCount: number;
  clickCount: number;
  createdAt: string;
  updatedAt: string;
  sentAt: string | null;
  createdBy: string;
}

export interface CommunicationHistoryEntry {
  id: string;
  campaignId: string;
  campaignName: string;
  type: CommunicationType;
  eventId: string;
  eventName: string;
  recipientCount: number;
  status: CommunicationStatus;
  sentAt: string | null;
  scheduledFor: string | null;
  subject: string;
  openRate: number;
  clickRate: number;
  createdBy: string;
}

export interface AIDraftRequest {
  type: CommunicationType;
  eventName: string;
  eventDate: string;
  eventVenue: string;
  tone: 'formal' | 'friendly' | 'urgent' | 'celebratory';
  additionalContext?: string;
  existingDraft?: string;
}

export interface AIDraftResponse {
  subject: string;
  bodyHtml: string;
  bodyText: string;
  suggestions: string[];
  confidence: number;
}

export interface CommunicationsState {
  campaigns: Campaign[];
  templates: CommunicationTemplate[];
  history: CommunicationHistoryEntry[];
  selectedCampaignId: string | null;
  composerOpen: boolean;
  composerMode: 'create' | 'edit';
  composerDraft: Partial<Campaign> | null;
  activeTab: 'campaigns' | 'history' | 'templates';
  filterType: CommunicationType | 'all';
  filterStatus: CommunicationStatus | 'all';
  isAILoading: boolean;
  aiError: string | null;
}
```
