// src/core/lifecycle/EventLifecycleEngine.ts

import { LifecycleEventBus } from './LifecycleEventBus';
import { LifecycleTransitionRules } from './LifecycleTransitionRules';
import { LifecycleStateStore } from './LifecycleStateStore';

export const LIFECYCLE_STATES = {
  PLANNING: 'planning',
  INVITING: 'inviting',
  LIVE: 'live',
  FOLLOW_UP: 'follow_up',
  INSIGHTS: 'insights',
} as const;

export type LifecycleState = typeof LIFECYCLE_STATES[keyof typeof LIFECYCLE_STATES];

export const LIFECYCLE_ORDER: LifecycleState[] = [
  'planning',
  'inviting',
  'live',
  'follow_up',
  'insights',
];

export interface LifecycleTransition {
  from: LifecycleState;
  to: LifecycleState;
  trigger: 'automatic' | 'manual';
  triggeredBy?: string; // userId for manual
  timestamp: string;
  reason?: string;
}

export interface EventLifecycleContext {
  eventId: string;
  currentState: LifecycleState;
  previousState: LifecycleState | null;
  transitionHistory: LifecycleTransition[];
  manualOverrideActive: boolean;
  overrideExpiresAt?: string;
  eventStartTime?: string;
  eventEndTime?: string;
  metadata: Record<string, unknown>;
}

export class EventLifecycleEngine {
  private store: LifecycleStateStore;
  private eventBus: LifecycleEventBus;
  private rules: LifecycleTransitionRules;
  private autoCheckInterval: ReturnType<typeof setInterval> | null = null;

  constructor(
    store: LifecycleStateStore,
    eventBus: LifecycleEventBus,
    rules: LifecycleTransitionRules
  ) {
    this.store = store;
    this.eventBus = eventBus;
    this.rules = rules;
  }

  /**
   * Initialize lifecycle for a new or existing event.
   * AC: Workspace initializes in Planning state for new events.
   */
  async initializeEvent(
    eventId: string,
    eventMeta: {
      startTime?: string;
      endTime?: string;
      existingState?: LifecycleState;
    }
  ): Promise<EventLifecycleContext> {
    const existingContext = await this.store.getContext(eventId);

    if (existingContext) {
      // Resume — rehydrate and validate current state is still correct
      const correctedState = await this.evaluateAutomaticTransition(existingContext);
      if (correctedState && correctedState !== existingContext.currentState) {
        return this.executeTransition(existingContext, correctedState, 'automatic');
      }
      return existingContext;
    }

    // New event — start in Planning
    const initialState: LifecycleState =
      eventMeta.existingState ?? LIFECYCLE_STATES.PLANNING;

    const context: EventLifecycleContext = {
      eventId,
      currentState: initialState,
      previousState: null,
      transitionHistory: [],
      manualOverrideActive: false,
      eventStartTime: eventMeta.startTime,
      eventEndTime: eventMeta.endTime,
      metadata: {},
    };

    await this.store.saveContext(context);
    this.eventBus.emit('lifecycle:initialized', { eventId, state: initialState });
    this.startAutomaticTransitionMonitor(eventId);

    return context;
  }

  /**
   * Manual override — allows host to force a specific state.
   * AC: Manual override must be explicitly confirmed and logged.
   */
  async manualOverride(
    eventId: string,
    targetState: LifecycleState,
    userId: string,
    reason: string,
    overrideDurationMinutes?: number
  ): Promise<EventLifecycleContext> {
    const context = await this.store.getContextOrThrow(eventId);

    if (context.currentState === targetState) {
      throw new Error(`Event is already in state: ${targetState}`);
    }

    const validStates = this.rules.getAllowedManualTargets(context.currentState);
    if (!validStates.includes(targetState)) {
      throw new Error(
        `Manual transition from ${context.currentState} to ${targetState} is not permitted. ` +
        `Allowed targets: ${validStates.join(', ')}`
      );
    }

    const updatedContext = await this.executeTransition(
      context,
      targetState,
      'manual',
      userId,
      reason
    );

    // Set override flag — prevents automatic transitions from overwriting
    updatedContext.manualOverrideActive = true;
    if (overrideDurationMinutes) {
      const expiry = new Date();
      expiry.setMinutes(expiry.getMinutes() + overrideDurationMinutes);
      updatedContext.overrideExpiresAt = expiry.toISOString();
    }

    await this.store.saveContext(updatedContext);
    this.eventBus.emit('lifecycle:manual_override', {
      eventId,
      from: context.currentState,
      to: targetState,
      userId,
      reason,
    });

    return updatedContext;
  }
