## 2. Data Model

### 2.1 Prisma Schema (`prisma/schema.prisma`)

```prisma
// Append to existing schema — no breaking changes to other models

enum GuestStatus {
  INVITED
  ACCEPTED
  DECLINED
  WAITLISTED
  REMOVED
}

model Guest {
  id             String      @id @default(cuid())
  eventId        String
  userId         String
  status         GuestStatus @default(INVITED)
  portfolioTags  String[]    // MVP seam — read/write, no business logic yet
  invitedAt      DateTime    @default(now())
  updatedAt      DateTime    @updatedAt

  event          Event       @relation(fields: [eventId], references: [id], onDelete: Cascade)

  @@unique([eventId, userId])           // one record per guest per event
  @@index([eventId])
  @@index([userId])
  @@index([status])
  @@map("guests")
}
```

## 3. Service Layer (`src/services/guestService.ts`)

```typescript
import { prisma } from '@/lib/prisma';
import type {
  Guest,
  GuestSummary,
  CreateGuestInput,
  UpdateGuestStatusInput,
} from '@/types/guest';

// ─── Read ────────────────────────────────────────────────────────────────────

export async function getGuestsByEvent(eventId: string): Promise<GuestSummary[]> {
  if (!eventId) throw new Error('eventId is required');

  return prisma.guest.findMany({
    where:   { eventId, status: { not: 'REMOVED' } },
    select:  { id: true, userId: true, status: true, portfolioTags: true },
    orderBy: { invitedAt: 'asc' },
  });
}

export async function getGuestById(guestId: string): Promise<Guest | null> {
  return prisma.guest.findUnique({ where: { id: guestId } });
}

// ─── Write ───────────────────────────────────────────────────────────────────

export async function createGuest(input: CreateGuestInput): Promise<Guest> {
  const { eventId, userId, portfolioTags = [] } = input;

  if (!eventId || !userId) {
    throw new Error('eventId and userId are required');
  }

  // Idempotent: if guest already exists, return existing record
  return prisma.guest.upsert({
    where:  { eventId_userId: { eventId, userId } },
    create: { eventId, userId, portfolioTags, status: 'INVITED' },
    update: {},                          // no-op if already exists
  });
}

export async function updateGuestStatus(
  input: UpdateGuestStatusInput,
): Promise<Guest> {
  const { guestId, status } = input;

  // REMOVED is a soft-delete — never hard-delete guests in MVP
  return prisma.guest.update({
    where: { id: guestId },
    data:  { status },
  });
}

// ─── Not implemented in MVP ───────────────────────────────────────────────────

// bulkInvite           → deferred (no integrations in MVP)
// exportGuestList      → deferred
```

## 4. API Routes

### 4.1 List guests (`src/app/api/events/[eventId]/guests/route.ts`)

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { getGuestsByEvent, createGuest } from '@/services/guestService';

// MVP auth stub — replace when TASK-0034 lands
function getRequestingUserId(req: NextRequest): string {
  return req.headers.get('x-stub-user-id') ?? 'anonymous';
}

export async function GET(
  _req: NextRequest,
  { params }: { params: { eventId: string } },
) {
  try {
    const guests = await getGuestsByEvent(params.eventId);
    return NextResponse.json({ guests });
  } catch (err) {
    console.error('[GET /guests]', err);
    return NextResponse.json({ error: 'Failed to fetch guests' }, { status: 500 });
  }
}

export async function POST(
  req: NextRequest,
  { params }: { params: { eventId: string } },
) {
  try {
    const body = await req.json();
    const userId = getRequestingUserId(req);

    const guest = await createGuest({
      eventId:       params.eventId,
      userId:        body.userId ?? userId,
      portfolioTags: body.portfolioTags ?? [],
    });

    return NextResponse.json({ guest }, { status: 201 });
  } catch (err) {
    console.error('[POST /guests]', err);
    return NextResponse.json({ error: 'Failed to create guest' }, { status: 500 });
  }
}
```
