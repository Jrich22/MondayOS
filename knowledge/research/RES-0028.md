export type GuestStatus =
  | 'invited'
  | 'confirmed'
  | 'declined'
  | 'waitlisted'
  | 'attended';

/**
 * Portfolio-tag seam: intentionally minimal for MVP.
 * Populated by a future portfolio integration task (not TASK-0021).
 * Shape is locked here so downstream tasks can depend on it.
 */
export interface PortfolioTagSeam {
  portfolioId: string | null;   // null until linked
  tagIds: string[];             // empty until portfolio integration ships
}

export interface Guest {
  id: string;                   // UUID, assigned at creation
  eventId: string;              // FK → Event.id
  displayName: string;
  email: string;
  status: GuestStatus;
  portfolioTag: PortfolioTagSeam;
  invitedAt: Date;
  updatedAt: Date;
}

/** DTO for adding a new guest — caller never sets id/timestamps. */
export interface CreateGuestInput {
  eventId: string;

  status?: GuestStatus;         // defaults to 'invited'
}

export interface UpdateGuestStatusInput {

## 2. Service Layer

### `src/services/guestService.ts`

```typescript
import { v4 as uuidv4 } from 'uuid';
import {
  Guest,
  CreateGuestInput,
  UpdateGuestStatusInput,
  GuestStatus,
} from '../models/Guest';

/**
 * In-process store for MVP (no production auth / DB — per AC).
 * Replaced by a real persistence layer in a post-MVP task.
 */
const store = new Map<string, Guest>();

// ─── Helpers ────────────────────────────────────────────────────────────────

function now(): Date {
  return new Date();
}

function guardEmail(email: string): void {
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    throw new Error(`Invalid email address: "${email}"`);
  }
}

// ─── Public API ─────────────────────────────────────────────────────────────

export function addGuest(input: CreateGuestInput): Guest {
  guardEmail(input.email);

  const duplicate = [...store.values()].find(
    (g) => g.eventId === input.eventId && g.email === input.email
  );
  if (duplicate) {
    throw new Error(
      `Guest with email "${input.email}" already exists on event "${input.eventId}".`
    );
  }

  const guest: Guest = {
    id: uuidv4(),
    eventId: input.eventId,
    displayName: input.displayName.trim(),
    email: input.email.toLowerCase().trim(),
    status: input.status ?? 'invited',
    portfolioTag: { portfolioId: null, tagIds: [] },
    invitedAt: now(),
    updatedAt: now(),
  };

  store.set(guest.id, guest);
  return guest;
}

export function updateGuestStatus(input: UpdateGuestStatusInput): Guest {
  const guest = store.get(input.guestId);
  if (!guest) {
    throw new Error(`Guest "${input.guestId}" not found.`);
  }

  const updated: Guest = {
    ...guest,
    status: input.status,
    updatedAt: now(),
  };

  store.set(updated.id, updated);
  return updated;
}

export function getGuestsForEvent(eventId: string): Guest[] {
  return [...store.values()]
    .filter((g) => g.eventId === eventId)
    .sort((a, b) => a.invitedAt.getTime() - b.invitedAt.getTime());
}

export function getGuestById(guestId: string): Guest | undefined {
  return store.get(guestId);
}

/** Test/demo utility — never exposed via API routes. */
export function _clearStore(): void {
  store.clear();
}
```

## 3. UI Component

### `src/components/GuestList.tsx`

```tsx
import React from 'react';
import { Guest, GuestStatus } from '../models/Guest';

// ─── Status Badge ────────────────────────────────────────────────────────────

const STATUS_STYLES: Record<GuestStatus, string> = {
  invited:    'bg-blue-100   text-blue-800',
  confirmed:  'bg-green-100  text-green-800',
  declined:   'bg-red-100    text-red-800',
  waitlisted: 'bg-yellow-100 text-yellow-800',
  attended:   'bg-purple-100 text-purple-800',
};

function StatusBadge({ status }: { status: GuestStatus }) {
  return (
    <span
      className={`inline-flex items-center rounded-full px-2 py-0.5 text-xs font-medium ${STATUS_STYLES[status]}`}
    >
      {status}
    </span>
  );
}

// ─── Guest Row ───────────────────────────────────────────────────────────────

function GuestRow({ guest }: { guest: Guest }) {
  return (
    <li className="flex items-center justify-between py-3 px-4 hover:bg-gray-50">
      <div className="flex flex-col">
        <span className="text-sm font-medium text-gray-900">
          {guest.displayName}
        </span>
        <span className="text-xs text-gray-500">{guest.email}</span>
      </div>
      <StatusBadge status={guest.status} />
    </li>
  );
}

// ─── Empty State ─────────────────────────────────────────────────────────────

function EmptyState() {
  return (
    <li className="py-8 text-center text-sm text-gray-400">
      No guests yet — add the first one.
    </li>
  );
}

// ─── GuestList ───────────────────────────────────────────────────────────────

interface GuestListProps {
  guests: Guest[];
  eventName?: string;
}

export function GuestList({ guests, eventName }: GuestListProps) {
  return (
    <section aria-label="Guest list">
      <div className="mb-3 flex items-center justify-between">
        <h2 className="text-base font-semibold text-gray-800">
          {eventName ? `${eventName} — Guests` : 'Guests'}
        </h2>
        <span className="text-xs text-gray-400">
          {guests.length} {guests.length === 1 ? 'guest' : 'guests'}
        </span>
      </div>

      <ul className="divide-y divide-gray-100 rounded-lg border border-gray-200 bg-white shadow-sm">
        {guests.length === 0
          ? <EmptyState />
          : guests.map((g) => <GuestRow key={g.id} guest={g} />)
        }
      </ul>
    </section>
  );
}

export default GuestList;
```
