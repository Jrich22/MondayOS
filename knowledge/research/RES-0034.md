/**
 * AI Assistant types — MVP placeholder surface.
 * No real generation, payment, integration, or production auth hooks.
 */

export type AIAssistantStatus = 'placeholder' | 'loading' | 'ready' | 'error';

export interface AIAssistantState {
  isOpen: boolean;
  status: AIAssistantStatus;
}

export interface AIAssistantContextValue {
  state: AIAssistantState;
  openAssistant: () => void;
  closeAssistant: () => void;
  toggleAssistant: () => void;
}

### `src/hooks/useAIAssistant.ts`

```typescript
import { useState, useCallback } from 'react';
import type { AIAssistantState, AIAssistantContextValue } from '../types/ai-assistant.types';

const INITIAL_STATE: AIAssistantState = {
  isOpen: false,
  status: 'placeholder',
};

/**
 * Hook managing AI assistant panel open/close state.
 *
 * MVP scope: status is always 'placeholder' — no real AI calls.
 * Future: status will drive loading/streaming states when backend is wired.
 */
export function useAIAssistant(): AIAssistantContextValue {
  const [state, setState] = useState<AIAssistantState>(INITIAL_STATE);

  const openAssistant = useCallback(() => {
    setState((prev) => ({ ...prev, isOpen: true }));
  }, []);

  const closeAssistant = useCallback(() => {
    setState((prev) => ({ ...prev, isOpen: false }));
  }, []);

  const toggleAssistant = useCallback(() => {
    setState((prev) => ({ ...prev, isOpen: !prev.isOpen }));
  }, []);

  return { state, openAssistant, closeAssistant, toggleAssistant };
}
```

### `src/hooks/useAIAssistant.test.ts`

```typescript
import { renderHook, act } from '@testing-library/react';
import { useAIAssistant } from './useAIAssistant';

describe('useAIAssistant', () => {
  it('initialises with panel closed and placeholder status', () => {
    const { result } = renderHook(() => useAIAssistant());
    expect(result.current.state.isOpen).toBe(false);
    expect(result.current.state.status).toBe('placeholder');
  });

  it('openAssistant sets isOpen to true', () => {
    const { result } = renderHook(() => useAIAssistant());
    act(() => result.current.openAssistant());
    expect(result.current.state.isOpen).toBe(true);
  });

  it('closeAssistant sets isOpen to false', () => {
    const { result } = renderHook(() => useAIAssistant());
    act(() => result.current.openAssistant());
    act(() => result.current.closeAssistant());
    expect(result.current.state.isOpen).toBe(false);
  });

  it('toggleAssistant flips isOpen state', () => {
    const { result } = renderHook(() => useAIAssistant());
    act(() => result.current.toggleAssistant());
    expect(result.current.state.isOpen).toBe(true);
    act(() => result.current.toggleAssistant());
    expect(result.current.state.isOpen).toBe(false);
  });

  it('status remains placeholder throughout — MVP scope enforcement', () => {
    const { result } = renderHook(() => useAIAssistant());
    act(() => result.current.openAssistant());
    act(() => result.current.closeAssistant());
    // Status must never deviate from 'placeholder' in MVP
    expect(result.current.state.status).toBe('placeholder');
  });

  it('openAssistant is stable across renders (memoised)', () => {
    const { result, rerender } = renderHook(() => useAIAssistant());
    const firstRef = result.current.openAssistant;
    rerender();
    expect(result.current.openAssistant).toBe(firstRef);
  });
});
```
