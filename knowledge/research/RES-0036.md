### `prisma/seed/helpers.ts`

```typescript
/**
 * Seed helpers — safe upserts, deterministic IDs, no production-auth calls.
 * All demo data uses a fixed namespace so seeds are idempotent.
 */

export const DEMO_NAMESPACE = "demo-seed-v1";

/** Stable fake IDs so re-runs produce the same rows */
export function demoId(domain: string, slug: string): string {
  return `${DEMO_NAMESPACE}:${domain}:${slug}`;
}

/** ISO date offset from "now" at seed time, keeps data feeling current */
export function daysFromNow(n: number): Date {
  const d = new Date();
  d.setDate(d.getDate() + n);
  d.setHours(19, 0, 0, 0); // 7 PM default event time
  return d;
}

export function daysAgo(n: number): Date {
  return daysFromNow(-n);
}
```

import { PrismaClient } from "@prisma/client";
import { demoId } from "./helpers";

/**
 * Demo users — NO production auth. Passwords are plaintext bcrypt-hashed

  {
    id: demoId("user", "admin"),

    name: "Alex Admin",
    role: "ADMIN",

    passwordHash:
      "$2b$10$K7L/8Y2qK1lJ3mN4oP5qR.uVwXyZ0aBcDeFgHiJkLmNoPqRsTuVwX",
    isDemo: true,
  },
  {
    id: demoId("user", "planner"),
    email: "planner@cue-demo.local",
    name: "Jordan Planner",
    role: "PLANNER",
    passwordHash:
      "$2b$10$K7L/8Y2qK1lJ3mN4oP5qR.uVwXyZ0aBcDeFgHiJkLmNoPqRsTuVwX",
    isDemo: true,
  },
] as const;

export async function seedUsers(prisma: PrismaClient) {
  console.log("  → Seeding demo users...");
  for (const user of DEMO_USERS) {
    await prisma.user.upsert({
      where: { email: user.email },
      update: {},      // idempotent — don't overwrite manual changes
      create: user,
    });
  }
  console.log(`     ✓ ${DEMO_USERS.length} users ready`);
}

export async function seedPortfolioTags(prisma: PrismaClient) {
  console.log("  → Seeding portfolio tags...");
  for (const tag of PORTFOLIO_TAGS) {

      where: { id: tag.id },
      update: { name: tag.name, category: tag.category },
      create: tag,
    });
  }
  console.log(`     ✓ ${PORTFOLIO_TAGS.length} tags ready`);
}
