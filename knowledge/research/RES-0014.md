  | 'live'
  | 'ended'
  | 'cancelled';

export interface Event {
  id: string;
  title: string;
  status: EventStatus;
  startsAt: string;   // ISO-8601
  endsAt: string;     // ISO-8601
  venue: string;
  capacity: number;
  ticketsSold: number;
  slug: string;
  createdAt: string;  // ISO-8601
  updatedAt: string;  // ISO-8601
}

  | 'status'

import type { EventStatus } from '@/types/event';

interface StatusConfig {
  label: string;
  /** Tailwind classes for the badge */
  badgeClasses: string;
  /** Numeric sort weight — lower = shown first */
  sortOrder: number;
  /** Accessible description for screen readers */
  description: string;
}

export const STATUS_CONFIG: Record<EventStatus, StatusConfig> = {
  live: {
    label: 'Live',
    badgeClasses: 'bg-green-100 text-green-800 ring-1 ring-green-600/20',
    sortOrder: 0,
    description: 'Event is currently in progress',
  },
  published: {
    label: 'Published',
    badgeClasses: 'bg-blue-100 text-blue-800 ring-1 ring-blue-600/20',
    sortOrder: 1,
    description: 'Event is published and accepting registrations',
  },
  draft: {
    label: 'Draft',
    badgeClasses: 'bg-gray-100 text-gray-700 ring-1 ring-gray-500/20',
    sortOrder: 2,
    description: 'Event is a draft and not yet visible to attendees',
  },
  ended: {
    label: 'Ended',
    badgeClasses: 'bg-yellow-100 text-yellow-800 ring-1 ring-yellow-600/20',
    sortOrder: 3,
    description: 'Event has concluded',
  },
  cancelled: {
    label: 'Cancelled',
    badgeClasses: 'bg-red-100 text-red-700 ring-1 ring-red-600/20',
    sortOrder: 4,
    description: 'Event has been cancelled',
  },
};

export function getSortedStatuses(): EventStatus[] {
  return (Object.keys(STATUS_CONFIG) as EventStatus[]).sort(
    (a, b) => STATUS_CONFIG[a].sortOrder - STATUS_CONFIG[b].sortOrder
  );
}
