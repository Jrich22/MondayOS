```typescript
export type FetchResult<T> =
  | { ok: true;  data: T }
  | { ok: false; error: string };

/**
 * Fetch a single event by ID.
 * Returns a typed result union — callers must handle the error branch.
 */
export async function getEventById(id: string): Promise<FetchResult<CueEvent>> {
  if (!id || typeof id !== 'string') {
    return { ok: false, error: 'Invalid event ID.' };
  }

  // Simulate network latency in development
  if (process.env.NODE_ENV === 'development') {
    await new Promise(resolve => setTimeout(resolve, 80));
  }

  const event = MOCK_EVENTS[id];
  if (!event) {
    return { ok: false, error: `Event "${id}" not found.` };
  }

  // Return a deep copy — prevent callers from mutating shared mock state
  return { ok: true, data: structuredClone(event) };
}
```

### 3. Utility: Date Formatting

```typescript
/**
 * Formats an ISO 8601 date string for display.
 * Returns a safe fallback rather than throwing — defensive UI pattern.
 */
export function formatEventDate(isoString: string): string {
  if (!isoString) return 'Date not set';

  try {
    const date = new Date(isoString);
    if (isNaN(date.getTime())) return 'Invalid date';

    return new Intl.DateTimeFormat('en-US', {
      weekday: 'long',
      year:    'numeric',
      month:   'long',
      day:     'numeric',
      hour:    '2-digit',
      minute:  '2-digit',
      timeZoneName: 'short',
    }).format(date);
  } catch {
    return 'Date unavailable';
  }
}
```
