    ACTIVE = "active"
    ON_HOLD = "on_hold"
    FILLED = "filled"
    CANCELLED = "cancelled"


@dataclass
class ReqWorkspace:
    """
    Primary workspace for a single open requisition.
    Aggregates all sourcing activity, candidates, and intelligence
    for one hiring need. GitHub issue/PR references are first-class.

    id: UUID = field(default_factory=uuid4)
    title: str = ""                          # e.g. "Senior Backend Engineer — Payments"

    status: ReqStatus = ReqStatus.DRAFT
    github_issue_url: Optional[str] = None   # Source of truth link

    updated_at: datetime = field(default_factory=datetime.utcnow)

    archived: bool = False
