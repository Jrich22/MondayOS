# TASK-0021: Guest List Model — Implementation

## Executive Summary

Implementing the guest/attendee data model for the Cue app MVP: a `Guest` entity with status tracking and a portfolio-tag seam, plus a basic guest list UI component rendered on an event.

---

## 1. Data Model Definition

### 1.1 Core Types (`src/types/guest.ts`)

```typescript
/**
 * Guest status lifecycle for MVP.
 * Deliberately minimal — no payment states (out of MVP scope).
 */
export type GuestStatus =
  | 'invited'
  | 'confirmed'
  | 'declined'
  | 'waitlisted'
  | 'attended';

/**
 * Portfolio-tag seam: a lightweight reference that can be
 * resolved later when the Portfolio domain is integrated.
 * Kept as a plain string ID so MVP has zero coupling to
 * that bounded context.
 */
export interface PortfolioTagSeam {
  portfolioId: string;       // opaque ref — resolved post-MVP
  label: string;             // human-readable snapshot at invite time
}

/**
 * Core guest entity.
 * No payments, no OAuth tokens, no external auth IDs — MVP scope only.
 */
export interface Guest {
  id: string;                          // uuid v4
  eventId: string;                     // FK → Event.id
  name: string;
  email: string;
  status: GuestStatus;
  portfolioTag?: PortfolioTagSeam;     // optional seam; null = general public guest
  note?: string;                       // organiser-only note
  invitedAt: string;                   // ISO-8601
  respondedAt?: string;                // ISO-8601; undefined until guest responds
  createdAt: string;
  updatedAt: string;
}

/**
 * DTO used when adding a new guest.
 * Omits server-assigned and derived fields.
 */
export type CreateGuestInput = Pick<Guest, 'eventId' | 'name' | 'email'> & {
  portfolioTag?: PortfolioTagSeam;
  note?: string;
};

/**
 * DTO used when updating guest status (RSVP flow).
 */
export type UpdateGuestStatusInput = {
  guestId: string;
  status: GuestStatus;
  respondedAt: string;
};
```

### 1.2 Design Decisions

| Decision | Rationale |
|---|---|
| `GuestStatus` as union string | Simple to serialise, easy to extend without migrations |
| `portfolioTag` is optional + opaque | Zero coupling to Portfolio domain; fulfils seam requirement without blocking MVP |
| No `userId` FK on `Guest` | Production auth is out of MVP scope; identity linked by email only |
| Separate `respondedAt` | Enables response-time analytics without parsing status history |

## 2. In-Memory Store (MVP persistence layer)

### `src/store/guestStore.ts`

```typescript
import { v4 as uuidv4 } from 'uuid';
import type {
  Guest,
  CreateGuestInput,
  UpdateGuestStatusInput,
} from '../types/guest';

// ---------------------------------------------------------------------------
// State — replaced by DB adapter post-MVP; interface is the contract.
// ---------------------------------------------------------------------------
let guests: Guest[] = [];

// ---------------------------------------------------------------------------
// CRUD helpers
// ---------------------------------------------------------------------------

export function getGuestsByEvent(eventId: string): Guest[] {
  return guests.filter((g) => g.eventId === eventId);
}

export function getGuestById(id: string): Guest | undefined {
  return guests.find((g) => g.id === id);
}

export function addGuest(input: CreateGuestInput): Guest {
  const now = new Date().toISOString();
  const guest: Guest = {
    id: uuidv4(),
    eventId: input.eventId,
    name: input.name,
    email: input.email,
    status: 'invited',
    portfolioTag: input.portfolioTag,
    note: input.note,
    invitedAt: now,
    createdAt: now,
    updatedAt: now,
  };
  guests = [...guests, guest];
  return guest;
}

export function updateGuestStatus(input: UpdateGuestStatusInput): Guest {
  const idx = guests.findIndex((g) => g.id === input.guestId);
  if (idx === -1) throw new Error(`Guest not found: ${input.guestId}`);

  const updated: Guest = {
    ...guests[idx],
    status: input.status,
    respondedAt: input.respondedAt,
    updatedAt: new Date().toISOString(),
  };
  guests = [...guests.slice(0, idx), updated, ...guests.slice(idx + 1)];
  return updated;
}

export function removeGuest(id: string): void {
  guests = guests.filter((g) => g.id !== id);
}

/** Test-only reset — keeps production code free of test concerns. */
export function __resetGuestStore(): void {
  guests = [];
}
```
