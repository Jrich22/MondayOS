## 3. API Layer

**`src/api/events.ts`**
```typescript
import { EventListParams, EventListResponse } from '../types/event';

const BASE_URL = process.env.REACT_APP_API_BASE_URL ?? '/api';

export async function fetchEvents(
  params: EventListParams = {}
): Promise<EventListResponse> {
  const {
    status = 'all',
    page = 1,
    pageSize = 20,
    sortBy = 'date',
    sortDir = 'asc',
  } = params;

  const query = new URLSearchParams({
    ...(status !== 'all' && { status }),
    page: String(page),
    pageSize: String(pageSize),
    sortBy,
    sortDir,
  });

  const res = await fetch(`${BASE_URL}/events?${query}`, {
    headers: { 'Content-Type': 'application/json' },
    credentials: 'include', // session cookie — no production auth in MVP
  });

  if (!res.ok) {
    const text = await res.text();
    throw new Error(`Failed to fetch events (${res.status}): ${text}`);
  }

  return res.json() as Promise<EventListResponse>;
}
```

## 4. Custom Hook

**`src/hooks/useEventList.ts`**
```typescript
import { useState, useEffect, useCallback } from 'react';
import { fetchEvents } from '../api/events';
import { Event, EventListParams, EventStatus } from '../types/event';

interface UseEventListState {
  events: Event[];
  total: number;
  page: number;
  isLoading: boolean;
  error: string | null;
}

interface UseEventListReturn extends UseEventListState {
  setPage: (page: number) => void;
  setStatusFilter: (status: EventStatus | 'all') => void;
  refresh: () => void;
}

const PAGE_SIZE = 20;

export function useEventList(): UseEventListReturn {
  const [params, setParams] = useState<EventListParams>({
    status: 'all',
    page: 1,
    pageSize: PAGE_SIZE,
    sortBy: 'date',
    sortDir: 'asc',
  });

  const [state, setState] = useState<UseEventListState>({
    events: [],
    total: 0,
    page: 1,
    isLoading: true,
    error: null,
  });

  const [refreshKey, setRefreshKey] = useState(0);

  useEffect(() => {
    let cancelled = false;

    setState(prev => ({ ...prev, isLoading: true, error: null }));

    fetchEvents(params)
      .then(data => {
        if (!cancelled) {
          setState({
            events: data.events,
            total: data.total,
            page: data.page,
            isLoading: false,
            error: null,
          });
        }
      })
      .catch((err: Error) => {
        if (!cancelled) {
          setState(prev => ({
            ...prev,
            isLoading: false,
            error: err.message,
          }));
        }
      });

    return () => { cancelled = true; };
  }, [params, refreshKey]);

  const setPage = useCallback((page: number) => {
    setParams(prev => ({ ...prev, page }));
  }, []);

  const setStatusFilter = useCallback((status: EventStatus | 'all') => {
    setParams(prev => ({ ...prev, status, page: 1 })); // reset to page 1
  }, []);

  const refresh = useCallback(() => {
    setRefreshKey(k => k + 1);
  }, []);

  return { ...state, setPage, setStatusFilter, refresh };
}
```

interface Props {

  cancelled: { label: 'Cancelled', className: 'badge--cancelled' },
  completed: { label: 'Completed', className: 'badge--completed' },
};

export const StatusBadge: React.FC<Props> = ({ status }) => {
  const { label, className } = CONFIG[status] ?? {
    label: status,
    className: 'badge--unknown',
  };

  return (
    <span
      className={`badge ${className}`}
      aria-label={`Event status: ${label}`}
    >
      {label}
    </span>
  );
};

import React from 'react';

interface Props {
  sold: number;
  total: number;
}

export const CapacityBar: React.FC<Props> = ({ sold, total }) => {
  const pct = total > 0 ? Math.min(100, Math.round((sold / total) * 100)) : 0;
  const label = `${sold.toLocaleString()} / ${total.toLocaleString()} tickets sold (${pct}%)`;

  return (
    <div className="capacity-bar" title={label} aria-label={label}>
      <div
        className="capacity-bar__fill"
        style={{ width: `${pct}%` }}
        role="presentation"
      />
      <span className="capacity-bar__label">{pct}%</span>
    </div>
  );
};
