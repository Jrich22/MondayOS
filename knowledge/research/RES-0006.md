# Auth Placeholder Implementation

## Analysis & Approach

Based on the objective, I need to create a fake current-user/session isolated behind a clean interface. The key design constraints are:

1. **Interface-first** — real auth drops in later without touching call sites
2. **No real auth** — no JWT verification, no OAuth, no password hashing
3. **MVP scope** — simple, not clever

### `src/auth/types.ts`

```typescript
/**
 * Represents an authenticated identity within the application.
 * Intentionally minimal — extend when real auth requirements are known.
 */
export interface CurrentUser {
  readonly id: string;
  readonly email: string;
  readonly displayName: string;
  readonly role: "admin" | "member";
}

/**
 * The seam that all application code depends on.
 * Real auth (Clerk, Auth0, Supabase, etc.) implements this interface.
 * The placeholder implements this interface.
 * Call sites never import a concrete class directly.
 */
export interface AuthProvider {
  /**
   * Returns the current user, or null if no session exists.
   * Intentionally synchronous for the placeholder; real implementations
   * may return Promise<CurrentUser | null> — adjust the interface when
   * that switchover happens.
   */
  getCurrentUser(): CurrentUser | null;

  /**
   * Whether any user is currently authenticated.
   * Derived from getCurrentUser() — kept explicit so call sites
   * read cleanly: `if (auth.isAuthenticated())` rather than
   * `if (auth.getCurrentUser() !== null)`.
   */
  isAuthenticated(): boolean;
}
```

### `src/auth/placeholder.ts`

```typescript
import type { AuthProvider, CurrentUser } from "./types";

/**
 * PLACEHOLDER ONLY — NOT FOR PRODUCTION USE.
 *
 * Simulates a hardcoded logged-in admin user so the rest of the
 * application can be built and tested without a real identity provider.
 *
 * Swap this out by creating a new class that implements AuthProvider
 * and updating the binding in index.ts. No other files change.
 */

const PLACEHOLDER_USER: CurrentUser = {
  id: "usr_placeholder_001",
  email: "dev@example.com",
  displayName: "Dev User (Placeholder)",
  role: "admin",
} as const;

export class PlaceholderAuthProvider implements AuthProvider {
  private readonly user: CurrentUser;

  constructor(overrides?: Partial<CurrentUser>) {
    // Allow tests to inject a different user without creating a new class.
    this.user = overrides ? { ...PLACEHOLDER_USER, ...overrides } : PLACEHOLDER_USER;
  }

  getCurrentUser(): CurrentUser | null {
    return this.user;
  }

  isAuthenticated(): boolean {
    return this.getCurrentUser() !== null;
  }
}
```

### `src/auth/__tests__/auth.test.ts`

```typescript
import { describe, it, expect } from "vitest";
import { PlaceholderAuthProvider } from "../placeholder";
import type { AuthProvider, CurrentUser } from "../types";

describe("PlaceholderAuthProvider", () => {
  it("satisfies the AuthProvider interface at compile time", () => {
    // If this compiles, the contract is met.
    const provider: AuthProvider = new PlaceholderAuthProvider();
    expect(provider).toBeDefined();
  });

  describe("getCurrentUser()", () => {
    it("returns a non-null user", () => {
      const auth = new PlaceholderAuthProvider();
      expect(auth.getCurrentUser()).not.toBeNull();
    });

    it("returns a user with all required fields populated", () => {
      const auth = new PlaceholderAuthProvider();
      const user = auth.getCurrentUser() as CurrentUser;

      expect(user.id).toBeTruthy();
      expect(user.email).toBeTruthy();
      expect(user.displayName).toBeTruthy();
      expect(user.role).toMatch(/^(admin|member)$/);
    });

    it("returns a stable reference on repeated calls", () => {
      const auth = new PlaceholderAuthProvider();
      expect(auth.getCurrentUser()).toBe(auth.getCurrentUser());
    });
  });

  describe("isAuthenticated()", () => {
    it("returns true for the default placeholder user", () => {
      const auth = new PlaceholderAuthProvider();
      expect(auth.isAuthenticated()).toBe(true);
    });
  });

  describe("constructor overrides", () => {
    it("allows role to be overridden for testing", () => {
      const auth = new PlaceholderAuthProvider({ role: "member" });
      expect(auth.getCurrentUser()?.role).toBe("member");
    });

    it("allows email to be overridden for testing", () => {
      const auth = new PlaceholderAuthProvider({ email: "test@test.com" });
      expect(auth.getCurrentUser()?.email).toBe("test@test.com");
    });

    it("preserves unoverridden fields", () => {
      const auth = new PlaceholderAuthProvider({ role: "member" });
      // id and displayName should still come from PLACEHOLDER_USER
      expect(auth.getCurrentUser()?.id).toBe("usr_placeholder_001");
    });
  });
});
```

// ✅ Correct — depend on the interface, import from the barrel

function WelcomeBanner() {
