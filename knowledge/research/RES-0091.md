# Cue App: Relationship Intelligence Foundation

## Executive Summary

I'll build the complete relationship intelligence foundation for Cue App — persistent Person profiles, relationship graphs, profile pages with timelines, AI summaries, and cross-event insights. All data is local/mock, no external dependencies.

## Implementation

### 1. Type Definitions

**`src/types/relationship.ts`**

```typescript
// ─────────────────────────────────────────────────────────────
// Core relationship intelligence types for Cue App
// Sprint 2: Local/mock data only — no production backend
// ─────────────────────────────────────────────────────────────

export type RelationshipStrength =
  | "strong"    // Regular contact, high mutual value
  | "active"    // Recent interaction, growing
  | "dormant"   // Was strong, needs nurturing
  | "new"       // First or second interaction
  | "weak";     // Infrequent, low signal

export type RelationshipContext =
  | "professional"
  | "investor"
  | "vendor"
  | "customer"
  | "collaborator"
  | "mentor"
  | "mentee"
  | "peer"
  | "personal";

export type CommunicationType =
  | "in_person_event"
  | "email"
  | "call"
  | "video_call"
  | "message"
  | "introduction"
  | "social_media";

export type OrganizationType =
  | "company"
  | "nonprofit"
  | "government"
  | "academic"
  | "community"
  | "other";

// ─────────────────────────────────────────────────────────────
// Person — the core persistent profile entity.
// Separate from Guest (event-scoped) — a Person persists
// across events and accumulates relational history.
// ─────────────────────────────────────────────────────────────
export interface Person {
  id: string;
  // Identity
  firstName: string;
  lastName: string;
  preferredName?: string;
  pronouns?: string;
  avatarUrl?: string;
  initials: string;        // Computed: FL

  // Professional
  title?: string;
  currentCompanyId?: string;
  previousCompanyIds: string[];
  linkedInUrl?: string;
  websiteUrl?: string;
  twitterHandle?: string;

  // Contact
  email?: string;
  phone?: string;
  location?: string;       // City, State

  // Relational metadata
  tags: string[];          // e.g. ["investor", "angel", "fintech"]
  notes: string;           // Free-form private notes
  aiSummary?: AISummary;

  // Graph connections
  relationshipIds: string[];  // edges in RelationshipGraph
  eventIds: string[];         // events attended together
  communicationIds: string[]; // interaction log entries

  // Lifecycle
  firstMetAt?: string;        // ISO date
  firstMetEventId?: string;
  createdAt: string;
  updatedAt: string;
  isArchived: boolean;

  // Sprint 2 flag — all data is mock
  _isMock: true;
}

// ─────────────────────────────────────────────────────────────
// Company / Organization
// ─────────────────────────────────────────────────────────────
export interface Company {
  id: string;
  name: string;
  type: OrganizationType;
  industry?: string;
  logoUrl?: string;
  websiteUrl?: string;
  location?: string;
  description?: string;
  size?: "1-10" | "11-50" | "51-200" | "201-1000" | "1000+";
  memberIds: string[];     // Person IDs currently at company
  alumniIds: string[];     // Person IDs who left
  tags: string[];
  createdAt: string;
  _isMock: true;
}

// ─────────────────────────────────────────────────────────────
// Relationship Edge — directed graph connection between entities
// ─────────────────────────────────────────────────────────────
export interface RelationshipEdge {
  id: string;
  fromPersonId: string;
  toPersonId: string;

  context: RelationshipContext;
  strength: RelationshipStrength;
  strengthScore: number;      // 0–100, computed from recency + frequency

  // How the relationship was established
  introducedByPersonId?: string;
  originEventId?: string;
  originDate: string;

  // Current state
  lastInteractionDate?: string;
  lastInteractionType?: CommunicationType;
  interactionCount: number;

  // Mutual connections (computed)
  mutualPersonIds: string[];

  notes?: string;
  isHidden: boolean;
  createdAt: string;
  updatedAt: string;
  _isMock: true;
}

// ─────────────────────────────────────────────────────────────
// Communication — a single interaction log entry
// ─────────────────────────────────────────────────────────────
export interface Communication {
  id: string;
  type: CommunicationType;
  date: string;             // ISO
  personIds: string[];      // All people involved
  eventId?: string;         // If tied to a Cue event
  subject?: string;
  summary?: string;
  sentiment?: "positive" | "neutral" | "negative";
  actionItemsGenerated: string[];
  followUpDate?: string;
  followUpCompleted: boolean;
  initiatedByUser: boolean;
  _isMock: true;
}

// ─────────────────────────────────────────────────────────────
// AI Summary — generated analysis of a person/relationship
// ─────────────────────────────────────────────────────────────
export interface AISummary {
  personId: string;
  headline: string;          // 1-line smart description
  professionalContext: string;
  relationshipInsight: string;
  talkingPoints: string[];   // Suggested conversation starters
  warnings: string[];        // e.g. "Hasn't responded in 60 days"
  opportunities: string[];   // e.g. "Could intro to Sarah Chen"
  generatedAt: string;
  modelVersion: string;      // For future real AI — tracked for invalidation
  _isMock: true;
}

// ─────────────────────────────────────────────────────────────
// Cross-Event Insight — pattern detected across multiple events
// ─────────────────────────────────────────────────────────────
export interface CrossEventInsight {
  id: string;
  type:
    | "recurring_attendee"    // Person attends same events as user
    | "cluster_detected"      // Group of people all know each other
    | "introduction_path"     // User can connect two people
    | "follow_up_overdue"     // No contact after meeting
    | "relationship_drift"    // Was strong, now dormant
    | "new_connection_spike"  // Met many new people at one event
