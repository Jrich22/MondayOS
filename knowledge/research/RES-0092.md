## Step 1: Core Type Definitions

**`src/types/relationship.ts`**

```typescript
// ============================================================
// CORE ENTITY TYPES
// ============================================================

export type EntityKind =
  | "person"
  | "company"
  | "organization"
  | "event"
  | "communication";

export type RelationshipStrength = "strong" | "moderate" | "weak" | "dormant";

export type RelationshipType =
  | "colleague"
  | "client"
  | "vendor"
  | "investor"
  | "mentor"
  | "mentee"
  | "friend"
  | "acquaintance"
  | "collaborator"
  | "competitor";

export type CommunicationType =
  | "email"
  | "meeting"
  | "call"
  | "message"
  | "social"
  | "event_interaction";

export type SentimentScore = -1 | -0.5 | 0 | 0.5 | 1;

// ============================================================
// PERSON PROFILE — persistent, separate from Guest records
// ============================================================

export interface SocialLinks {
  linkedin?: string;
  twitter?: string;
  github?: string;
  website?: string;
}

export interface ContactInfo {
  email?: string;
  phone?: string;
  location?: string;
  timezone?: string;
}

export interface PersonProfile {
  id: string;                          // uuid, not tied to guest record
  guestRecordIds: string[];            // links to zero-or-more Guest records
  createdAt: string;                   // ISO 8601
  updatedAt: string;

  // Identity
  firstName: string;
  lastName: string;
  preferredName?: string;
  avatarUrl?: string;
  headline?: string;                   // "VP Engineering @ Acme"

  // Affiliation
  currentCompanyId?: string;
  currentTitle?: string;
  previousCompanyIds: string[];

  // Contact
  contact: ContactInfo;
  social: SocialLinks;

  // Tags & classification
  tags: string[];                      // e.g. ["investor", "speaker", "warm-lead"]
  relationshipType: RelationshipType;
  strength: RelationshipStrength;

  // AI-generated fields
  aiSummary?: AISummary;
  aiRecommendations: AIRecommendation[];

  // Computed / cached
  lastInteractionAt?: string;
  interactionCount: number;
  mutualConnectionIds: string[];
}

// ============================================================
// COMPANY / ORGANIZATION
// ============================================================

export interface Company {
  id: string;
  kind: "company" | "organization";
  name: string;
  domain?: string;
  logoUrl?: string;
  industry?: string;
  size?: "1-10" | "11-50" | "51-200" | "201-500" | "500+";
  description?: string;
  linkedPersonIds: string[];
  linkedEventIds: string[];
  tags: string[];
}

// ============================================================
// RELATIONSHIP EDGE — the graph connection
// ============================================================

export interface RelationshipEdge {
  id: string;
  sourceId: string;                    // PersonProfile.id
  targetId: string;                    // PersonProfile.id | Company.id | Event.id
  targetKind: EntityKind;
  type: RelationshipType;
  strength: RelationshipStrength;
  since?: string;                      // ISO date
  notes?: string;
  sharedEventIds: string[];
  communicationIds: string[];
  sentiment: SentimentScore;
  createdAt: string;
  updatedAt: string;
}

// ============================================================
// COMMUNICATION RECORD
// ============================================================

export interface CommunicationRecord {
  id: string;
  type: CommunicationType;
  participantIds: string[];            // PersonProfile.ids
  eventId?: string;
  subject?: string;
  summary?: string;
  sentiment: SentimentScore;
  occurredAt: string;                  // ISO 8601
  durationMinutes?: number;
  topics: string[];
  actionItems: string[];
  attachmentUrls: string[];
}

// ============================================================
// EVENT LINK (to Cue events)
// ============================================================

export interface EventParticipation {
  eventId: string;
  eventName: string;
  eventDate: string;
  role: "attendee" | "speaker" | "organizer" | "sponsor" | "exhibitor";
  personIds: string[];                 // who was present
  communicationIds: string[];
  notes?: string;
}

// ============================================================
// AI TYPES
// ============================================================

export interface AISummary {
  id: string;
  personId: string;
  generatedAt: string;
  model: string;                       // e.g. "gpt-4o-mini-mock"
  executiveSummary: string;            // 2–3 sentence overview
  strengths: string[];
  sharedInterests: string[];
  conversationStarters: string[];
  riskFlags: string[];                 // e.g. "No contact in 6 months"
  nextBestAction: string;
}

export interface AIRecommendation {
  id: string;
  personId: string;
  generatedAt: string;
  category: "reconnect" | "introduce" | "follow_up" | "nurture" | "escalate";
  priority: "high" | "medium" | "low";
  title: string;
  rationale: string;
  suggestedAction: string;
  dueBy?: string;
  dismissed: boolean;
}

// ============================================================
// CROSS-EVENT INSIGHT
// ============================================================

export interface CrossEventInsight {
  id: string;
  personId: string;
  eventIds: string[];
  insightType:
    | "pattern"
    | "introduction_opportunity"
    | "recurring_theme"
    | "momentum";
  title: string;
  description: string;
  confidence: number;                  // 0–1
  surfacedAt: string;
}

// ============================================================
// GRAPH SNAPSHOT (serialisable for rendering)
// ============================================================

export interface GraphNode {
  id: string;
  kind: EntityKind;
  label: string;
  strength?: RelationshipStrength;
  x?: number;
  y?: number;
  metadata: Record<string, unknown>;
}

export interface GraphEdge {
  id: string;
  source: string;
  target: string;
  type: RelationshipType;
  strength: RelationshipStrength;
  weight: number;                      // 1 (weak) → 4 (strong)
}

export interface RelationshipGraph {
  nodes: GraphNode[];
  edges: GraphEdge[];
  focusPersonId: string;
  generatedAt: string;
}

// ============================================================
// TIMELINE EVENT (unified interaction history)
// ============================================================

export interface TimelineEvent {
  id: string;
  personId: string;
  occurredAt: string;
  kind:
    | "first_met"
    | "event_attendance"
    | "communication"
    | "introduction"
    | "milestone"
    | "ai_note";
