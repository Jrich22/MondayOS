# TASK-0020: Create-Event Flow — Engineering Implementation

## Executive Summary

This delivers the complete `create-event` flow for the Cue App MVP: a multi-step form that collects event details, provisions a workspace, and persists a canonical `Event` record. All work is scoped to MVP (no payments, no external integrations, no production auth).

## 1. Domain Model

### `Event` Record (canonical schema)

```typescript
// packages/core/src/domain/event.ts

export type EventStatus = 'draft' | 'published' | 'archived';

export interface Event {
  id: string;                    // uuid v4
  workspaceId: string;           // FK → Workspace.id
  ownerId: string;               // FK → User.id (session user)
  title: string;                 // 3–120 chars
  description: string | null;    // max 2000 chars
  startsAt: Date;                // ISO 8601, must be future
  endsAt: Date;                  // must be > startsAt
  timezone: string;              // IANA tz string e.g. 'America/New_York'
  location: EventLocation | null;
  capacity: number | null;       // positive int, null = unlimited
  status: EventStatus;
  createdAt: Date;
  updatedAt: Date;
}

export interface EventLocation {
  type: 'physical' | 'virtual' | 'hybrid';
  address: string | null;        // for physical/hybrid
  virtualUrl: string | null;     // for virtual/hybrid
}

export interface Workspace {
  id: string;
  eventId: string;               // 1-to-1 with Event
  slug: string;                  // url-safe, derived from event title
  createdAt: Date;
}
```

```typescript
import { z } from 'zod';

const futureDate = z.coerce.date().refine(
  (d) => d > new Date(),
  { message: 'Date must be in the future' }
);

export const createEventSchema = z.object({
  title: z
    .string()
    .min(3, 'Title must be at least 3 characters')
    .max(120, 'Title must be 120 characters or fewer'),
  description: z.string().max(2000).nullable().default(null),
  startsAt: futureDate,
  endsAt: z.coerce.date(),
  timezone: z.string().min(1, 'Timezone is required'),
  location: z
    .object({
      type: z.enum(['physical', 'virtual', 'hybrid']),
      address: z.string().nullable().default(null),
      virtualUrl: z.string().url().nullable().default(null),
    })
    .nullable()
    .default(null),
  capacity: z.number().int().positive().nullable().default(null),
}).refine(
  (data) => data.endsAt > data.startsAt,
  { message: 'End time must be after start time', path: ['endsAt'] }
);

export type CreateEventInput = z.infer<typeof createEventSchema>;
```

## 2. Service Layer

### Event Service

```typescript
// packages/core/src/services/event.service.ts
import { v4 as uuid } from 'uuid';
import { createEventSchema, CreateEventInput } from '../domain/event.schema';
import { Event, Workspace } from '../domain/event';
import { EventRepository } from '../repositories/event.repository';
import { WorkspaceRepository } from '../repositories/workspace.repository';
import { slugify } from '../utils/slugify';

export class EventService {
  constructor(
    private readonly eventRepo: EventRepository,
    private readonly workspaceRepo: WorkspaceRepository,
  ) {}

  async createEvent(
    rawInput: unknown,
    ownerId: string,
  ): Promise<{ event: Event; workspace: Workspace }> {
    // 1. Validate input
    const input: CreateEventInput = createEventSchema.parse(rawInput);

    // 2. Generate IDs up-front (enables atomic rollback if needed)
    const eventId = uuid();
    const workspaceId = uuid();

    // 3. Derive workspace slug (unique-safe: append short id fragment)
    const baseSlug = slugify(input.title);
    const slug = `${baseSlug}-${eventId.slice(0, 8)}`;

    const now = new Date();

    const event: Event = {
      id: eventId,
      workspaceId,
      ownerId,
      title: input.title,
      description: input.description,
      startsAt: input.startsAt,
      endsAt: input.endsAt,
      timezone: input.timezone,
      location: input.location,
      capacity: input.capacity,
      status: 'draft',          // MVP: always starts as draft
      createdAt: now,
      updatedAt: now,
    };

    const workspace: Workspace = {
      id: workspaceId,
      eventId,
      slug,
      createdAt: now,
    };

    // 4. Persist — workspace first (event has FK to workspace)
    await this.workspaceRepo.create(workspace);
    await this.eventRepo.create(event);

    return { event, workspace };
  }
}
```

### In-Memory Implementations (MVP — no production DB required)

```typescript
// packages/core/src/repositories/in-memory/event.repository.ts
import { Event } from '../../domain/event';
import { EventRepository } from '../event.repository';

export class InMemoryEventRepository implements EventRepository {
  private store = new Map<string, Event>();

  async create(event: Event): Promise<void> {
    if (this.store.has(event.id)) {
      throw new Error(`Event ${event.id} already exists`);
    }
    this.store.set(event.id, { ...event });
  }

  async findById(id: string): Promise<Event | null> {
    return this.store.get(id) ?? null;
  }
}

// packages/core/src/repositories/in-memory/workspace.repository.ts
import { Workspace } from '../../domain/event';
import { WorkspaceRepository } from '../workspace.repository';

export class InMemoryWorkspaceRepository implements WorkspaceRepository {
  private store = new Map<string, Workspace>();

  async create(workspace: Workspace): Promise<void> {
    this.store.set(workspace.id, { ...workspace });
  }

  async findBySlug(slug: string): Promise<Workspace | null> {
    return [...this.store.values()].find((w) => w.slug === slug) ?? null;
  }
}
```
