## 1. Core Lifecycle State Machine

```typescript
// src/features/cue/lifecycle/CueLifecycleEngine.ts

import { EventEmitter } from 'events';

export type LifecycleState =
  | 'planning'
  | 'inviting'
  | 'live'
  | 'follow_up'
  | 'insights';

export type LifecycleTransitionTrigger =
  | 'manual_override'
  | 'schedule_reached'
  | 'rsvp_threshold_met'
  | 'event_started'
  | 'event_ended'
  | 'follow_up_period_elapsed';

export interface LifecycleTransition {
  from: LifecycleState;
  to: LifecycleState;
  trigger: LifecycleTransitionTrigger;
  timestamp: Date;
  actorId?: string; // userId if manual_override
  metadata?: Record<string, unknown>;
}

export interface CueEvent {
  id: string;
  title: string;
  scheduledStart: Date;
  scheduledEnd: Date;
  followUpWindowHours: number; // default 48
  rsvpThreshold: number; // % of invitees needed to auto-advance to 'live' readiness
  currentState: LifecycleState;
  stateHistory: LifecycleTransition[];
  isManualOverrideActive: boolean;
  overrideExpiresAt?: Date;
}

// Valid state transitions — directed graph
const VALID_TRANSITIONS: Record<LifecycleState, LifecycleState[]> = {
  planning: ['inviting'],
  inviting: ['planning', 'live'],   // can revert to planning
  live: ['follow_up'],
  follow_up: ['insights'],
  insights: [],                      // terminal state
};

const AUTO_TRANSITION_RULES: Array<{
  from: LifecycleState;
  to: LifecycleState;
  trigger: LifecycleTransitionTrigger;
  condition: (event: CueEvent, now: Date) => boolean;
}> = [
  {
    from: 'inviting',
    to: 'live',
    trigger: 'event_started',
    condition: (event, now) => now >= event.scheduledStart,
  },
  {
    from: 'live',
    to: 'follow_up',
    trigger: 'event_ended',
    condition: (event, now) => now >= event.scheduledEnd,
  },
  {
    from: 'follow_up',
    to: 'insights',
    trigger: 'follow_up_period_elapsed',
    condition: (event, now) => {
      const followUpEnd = new Date(event.scheduledEnd);
      followUpEnd.setHours(
        followUpEnd.getHours() + event.followUpWindowHours
      );
      return now >= followUpEnd;
    },
  },
];

export class CueLifecycleEngine extends EventEmitter {
  private tickIntervalMs: number;
  private tickHandle?: ReturnType<typeof setInterval>;

  constructor(tickIntervalMs = 60_000) {
    super();
    this.tickIntervalMs = tickIntervalMs;
  }

  // ── Public API ──────────────────────────────────────────────────────────────

  transition(
    event: CueEvent,
    to: LifecycleState,
    trigger: LifecycleTransitionTrigger,
    actorId?: string,
    metadata?: Record<string, unknown>
  ): CueEvent {
    const from = event.currentState;

    if (!this.isValidTransition(from, to)) {
      throw new LifecycleTransitionError(
        `Invalid transition: ${from} → ${to}`
      );
    }

    const transition: LifecycleTransition = {
      from,
      to,
      trigger,
      timestamp: new Date(),
      actorId,
      metadata,
    };

    const updated: CueEvent = {
      ...event,
      currentState: to,
      stateHistory: [...event.stateHistory, transition],
      isManualOverrideActive: trigger === 'manual_override',
      overrideExpiresAt:
        trigger === 'manual_override'
          ? this.computeOverrideExpiry(to, event)
          : undefined,
    };

    this.emit('state:changed', { event: updated, transition });
    return updated;
  }

  manualOverride(
    event: CueEvent,
    to: LifecycleState,
    actorId: string
  ): CueEvent {
    return this.transition(event, to, 'manual_override', actorId, {
      previousState: event.currentState,
    });
  }

  /**
   * Evaluate auto-transitions for a given event.
   * Returns updated event if a transition occurred, or the same object.
   */
  evaluateAutoTransitions(event: CueEvent, now: Date = new Date()): CueEvent {
    // Respect manual override — suppress auto-transitions until expiry
    if (
      event.isManualOverrideActive &&
      event.overrideExpiresAt &&
      now < event.overrideExpiresAt
    ) {
      return event;
    }

    for (const rule of AUTO_TRANSITION_RULES) {
      if (
        event.currentState === rule.from &&
        rule.condition(event, now)
      ) {
        return this.transition(event, rule.to, rule.trigger);
      }
    }

    return event;
  }

  /** Start background polling for a set of events */
  startAutoTransitionPoller(
    getEvents: () => Promise<CueEvent[]>,
    persistEvent: (event: CueEvent) => Promise<void>
  ): void {
    if (this.tickHandle) return; // already running

    this.tickHandle = setInterval(async () => {
      const events = await getEvents();
      const now = new Date();

      for (const event of events) {
        const updated = this.evaluateAutoTransitions(event, now);
        if (updated !== event) {
          await persistEvent(updated);
          this.emit('auto:transitioned', updated);
        }
      }
    }, this.tickIntervalMs);
  }

  stopAutoTransitionPoller(): void {
    if (this.tickHandle) {
      clearInterval(this.tickHandle);
      this.tickHandle = undefined;
    }
  }

  isValidTransition(from: LifecycleState, to: LifecycleState): boolean {
    return VALID_TRANSITIONS[from].includes(to);
  }

  getAvailableTransitions(state: LifecycleState): LifecycleState[] {
    return VALID_TRANSITIONS[state];
  }

  // ── Private helpers ─────────────────────────────────────────────────────────

  private computeOverrideExpiry(
    targetState: LifecycleState,
    event: CueEvent
  ): Date {
    // Override expires when the natural next trigger would fire
    const expiry = new Date();
    switch (targetState) {
      case 'live':
        return event.scheduledEnd;
      case 'follow_up': {
        const end = new Date(event.scheduledEnd);
        end.setHours(end.getHours() + event.followUpWindowHours);
        return end;
      }
      default:
        // 24-hour grace for planning / inviting overrides
        expiry.setHours(expiry.getHours() + 24);
        return expiry;
    }
  }
}
