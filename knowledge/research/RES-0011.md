### 3. Event Row Component

```typescript
// src/components/EventRow/EventRow.tsx
import React from 'react';
import type { CueEvent } from '../../types/event';
import { EventStatusBadge } from '../EventStatusBadge/EventStatusBadge';
import styles from './EventRow.module.css';

interface Props {
  event: CueEvent;
  onClick: (id: string) => void;
}

export const EventRow: React.FC<Props> = ({ event, onClick }) => {
  const eventDate = new Date(event.date);
  const soldPercent =
    event.capacity > 0
      ? Math.round((event.ticketsSold / event.capacity) * 100)
      : 0;

  return (
    <li
      className={styles.row}
      onClick={() => onClick(event.id)}
      data-testid={`event-row-${event.id}`}
      role="button"
      tabIndex={0}
      onKeyDown={(e) => e.key === 'Enter' && onClick(event.id)}
      aria-label={`Event: ${event.title}`}
    >
      <div className={styles.primary}>
        <span className={styles.title}>{event.title}</span>
        <EventStatusBadge status={event.status} />
      </div>
      <div className={styles.meta}>
        <span className={styles.date}>
          {eventDate.toLocaleDateString('en-GB', {
            day: 'numeric',
            month: 'short',
            year: 'numeric',
          })}
        </span>
        <span className={styles.venue}>{event.venue}</span>
        <span className={styles.capacity} aria-label="Tickets sold">
          {event.ticketsSold} / {event.capacity}
          <span className={styles.capacityPct}>({soldPercent}%)</span>
        </span>
      </div>
    </li>
  );
};
```
