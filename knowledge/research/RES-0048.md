/**
 * Core event types for the Cue App MVP.
 * Scoped to Sprint 1 — no payment, integration, or auth fields.
 */

export type EventStatus =
  | 'draft'
  | 'published'
  | 'in_progress'
  | 'completed'
  | 'cancelled';

export interface CueEvent {
  id: string;
  title: string;
  description: string;
  status: EventStatus;
  startDate: string;        // ISO 8601
  endDate: string;          // ISO 8601
  location: string;
  attendeeCount: number;
  capacity: number;
  createdAt: string;        // ISO 8601
  updatedAt: string;        // ISO 8601
}

export interface EventDashboardState {
  events: CueEvent[];
  isLoading: boolean;
  error: string | null;
  selectedStatus: EventStatus | 'all';
}
