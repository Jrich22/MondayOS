const MAX_NAME_LENGTH  = 120;
const MAX_NOTES_LENGTH = 1000;

export interface CreateGuestInput {
  eventId:      string;
  displayName:  string;
  email:        string;
  plusOnes?:    number;
  notes?:       string;
  portfolioTag?: PortfolioTagSeam;

export interface ValidationError {
  field:   string;
  message: string;
}

// ─── Validation ───────────────────────────────────────────────────────────────

export function validateGuestInput(
  input: CreateGuestInput
): ValidationError[] {
  const errors: ValidationError[] = [];

  if (!input.eventId?.trim()) {
    errors.push({ field: 'eventId', message: 'eventId is required.' });
  }

  if (!input.displayName?.trim()) {
    errors.push({ field: 'displayName', message: 'Display name is required.' });
  } else if (input.displayName.length > MAX_NAME_LENGTH) {
    errors.push({
      field: 'displayName',
      message: `Display name must be ≤ ${MAX_NAME_LENGTH} characters.`,
    });
  }

  if (!input.email?.trim()) {
    errors.push({ field: 'email', message: 'Email is required.' });
  } else if (!EMAIL_RE.test(input.email)) {
    errors.push({ field: 'email', message: 'Email format is invalid.' });
  }

  const plusOnes = input.plusOnes ?? 0;
  if (!Number.isInteger(plusOnes) || plusOnes < 0 || plusOnes > MAX_PLUS_ONES) {
    errors.push({
      field: 'plusOnes',
      message: `Plus-ones must be an integer between 0 and ${MAX_PLUS_ONES}.`,
    });
  }

  if (input.notes && input.notes.length > MAX_NOTES_LENGTH) {
    errors.push({
      field: 'notes',
      message: `Notes must be ≤ ${MAX_NOTES_LENGTH} characters.`,
    });
  }

  return errors;
}

// ─── Factory ─────────────────────────────────────────────────────────────────

export function createGuest(input: CreateGuestInput): Guest {
  const errors = validateGuestInput(input);
  if (errors.length > 0) {
    throw new Error(
      `Invalid guest input: ${errors.map(e => `${e.field} — ${e.message}`).join('; ')}`
    );
  }

  const now = new Date().toISOString();
  const initialStatus = GuestStatus.INVITED;

  return {
    id:          uuidv4(),
    eventId:     input.eventId.trim(),
    displayName: input.displayName.trim(),
    email:       input.email.trim().toLowerCase(),
    status:      initialStatus,
    statusHistory: [
      { status: initialStatus, changedAt: now, changedBy: 'system' },
    ],
    portfolioTag: input.portfolioTag,
    notes:        input.notes?.trim(),
    plusOnes:     input.plusOnes ?? 0,
    createdAt:    now,
    updatedAt:    now,
  };
}
