### Scope Evolution Strategy

The dashboard uses a **scope context** pattern. Phase 1 delivers single-event view. The data contracts and component APIs are designed from day one to accept multi-event and org-level data, so Phases 2 and 3 are additive, not rewrites.

## 2. Data Contracts (TypeScript)

```typescript
// src/types/mission-control.ts

export type DashboardScope = 'event' | 'organization' | 'portfolio';

export interface DashboardScopeContext {
  scope: DashboardScope;
  eventId?: string;          // populated when scope === 'event'
  organizationId: string;    // always present
  portfolioId?: string;      // populated when scope === 'portfolio'
}

// ─── Event Health ────────────────────────────────────────────────

export type HealthStatus = 'healthy' | 'warning' | 'critical' | 'unknown';

export interface EventHealthSignal {
  category: 'venue' | 'logistics' | 'attendance' | 'finance' | 'vendor' | 'comms';
  status: HealthStatus;
  label: string;
  detail: string;
  lastUpdatedAt: string; // ISO-8601
  actionable: boolean;
}

export interface EventHealthSummary {
  eventId: string;
  eventName: string;
  eventDate: string;       // ISO-8601
  overallStatus: HealthStatus;
  signals: EventHealthSignal[];
  computedAt: string;
}

// ─── Attendance ──────────────────────────────────────────────────

export interface AttendanceSnapshot {
  eventId: string;
  capacity: number;
  registered: number;
  checkedIn: number;        // live, real-time
  waitlisted: number;
  cancellations: number;
  registrationVelocity: number;  // registrations per hour, trailing 2h
  projectedFinalAttendance: number;
  capacityUtilizationPct: number;
  trend: 'accelerating' | 'steady' | 'decelerating' | 'stalled';
  asOf: string;
}

// ─── Alerts ──────────────────────────────────────────────────────

export type AlertSeverity = 'info' | 'warning' | 'critical';
export type AlertCategory =
  | 'attendance'
  | 'logistics'
  | 'vendor'
  | 'finance'
  | 'weather'
  | 'system'
  | 'comms';

export interface Alert {
  id: string;
  severity: AlertSeverity;
  category: AlertCategory;
  title: string;
  body: string;
  eventId?: string;
  createdAt: string;
  acknowledgedAt?: string;
  resolvedAt?: string;
  suggestedActionId?: string; // links to an AI recommendation
}

// ─── AI Recommendations ──────────────────────────────────────────

export type RecommendationConfidence = 'low' | 'medium' | 'high';
export type RecommendationStatus = 'pending' | 'accepted' | 'dismissed' | 'snoozed';

export interface AIRecommendation {
  id: string;
  eventId?: string;
  title: string;
  rationale: string;
  suggestedAction: string;
  confidence: RecommendationConfidence;
  estimatedImpact: string;   // human-readable, e.g. "Reduces no-shows by ~12%"
  linkedAlertIds: string[];
  status: RecommendationStatus;
  expiresAt?: string;
  createdAt: string;
}

// ─── Milestones ──────────────────────────────────────────────────

export type MilestoneStatus = 'not_started' | 'in_progress' | 'complete' | 'at_risk' | 'overdue';

export interface Milestone {
  id: string;
  eventId: string;
  title: string;
  dueDate: string;
  status: MilestoneStatus;
  ownerName: string;
  ownerAvatarUrl?: string;
  completionPct: number;
  blockers: string[];
  isUpcoming: boolean;      // within next 14 days
}

// ─── Org Metrics ─────────────────────────────────────────────────

export interface OrgMetric {
  key: string;
  label: string;
  value: number | string;
  unit?: string;
  changeVsPriorPeriod?: number;  // signed percentage
  trend: 'up' | 'down' | 'flat';
  goodDirection: 'up' | 'down';  // for coloring — e.g. cancellations: 'down'
}

export interface OrgMetricsSummary {
  organizationId: string;
  periodLabel: string;           // e.g. "This Quarter"
  metrics: OrgMetric[];
  asOf: string;
}

// ─── Root dashboard payload ───────────────────────────────────────

export interface MissionControlPayload {
  scope: DashboardScopeContext;
  eventHealth: EventHealthSummary[];       // 1 for event scope, N for org/portfolio
  attendance: AttendanceSnapshot[];
  alerts: Alert[];
  recommendations: AIRecommendation[];
  milestones: Milestone[];
  orgMetrics?: OrgMetricsSummary;
}
```
