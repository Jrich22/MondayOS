# AI Assistant Placeholder — Engineering Delivery

## Overview

This delivers the AI assistant placeholder surface for the Cue app (TASK-0029, Epic: ai_layer). The implementation is intentionally scoped to MVP: a visible, interactive entry point that communicates future AI planning capability without wiring any real generation, payments, integrations, or production auth.

---

## Implementation Plan

### Architecture Decision

The placeholder lives as a standalone panel/drawer component (`AiAssistantPanel`) rendered within the existing app shell. It is:
- **Discoverable** — triggered by a persistent FAB or sidebar nav item
- **Honest** — clearly labeled "Coming Soon" / "Preview" so users understand it's not yet functional
- **Extensible** — the component interface is designed so real AI calls slot in without structural refactoring

import React, { useState } from 'react';
import { X, Sparkles, Send, Lock } from 'lucide-react';
import clsx from 'clsx';

export interface AiAssistantPanelProps {
  /** Whether the panel is visible */
  isOpen: boolean;
  /** Called when the user closes the panel */
  onClose: () => void;
  /**
   * Future: inject a real handler once AI layer (TASK-0030) is ready.
   * Placeholder always returns a canned response.
   */
  onSubmit?: (prompt: string) => Promise<string>;
}

interface Message {
  id: string;
  role: 'user' | 'assistant';
  content: string;
  isPlaceholder?: boolean;
}

const PLACEHOLDER_RESPONSE =
  "AI planning help is coming soon! Once available, I'll help you break down goals, schedule tasks, and stay on track.";

const SUGGESTED_PROMPTS = [
  'Help me plan my week',
  'Break this goal into tasks',
  'What should I focus on today?',
];

export function AiAssistantPanel({
  isOpen,
  onClose,
  onSubmit,
}: AiAssistantPanelProps) {
  const [input, setInput] = useState('');
  const [messages, setMessages] = useState<Message[]>([
    {
      id: 'welcome',
      role: 'assistant',
      content:
        "Hi! I'm your AI planning assistant. I'm not quite ready yet, but you can explore what I'll be able to help with.",
      isPlaceholder: true,
    },
  ]);
  const [isLoading, setIsLoading] = useState(false);

  const handleSubmit = async (prompt: string) => {
    if (!prompt.trim()) return;

    const userMessage: Message = {
      id: `user-${Date.now()}`,
      role: 'user',
      content: prompt.trim(),
    };

    setMessages((prev) => [...prev, userMessage]);
    setInput('');
    setIsLoading(true);

    try {
      // If a real handler is injected (future sprint), use it.
      // Otherwise fall back to the placeholder response.
      const response = onSubmit
        ? await onSubmit(prompt.trim())
        : await simulatePlaceholderResponse();

      const assistantMessage: Message = {
        id: `assistant-${Date.now()}`,
        role: 'assistant',
        content: response,
        isPlaceholder: !onSubmit,
      };

      setMessages((prev) => [...prev, assistantMessage]);
    } finally {
      setIsLoading(false);
    }
  };

  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      handleSubmit(input);
    }
  };

  if (!isOpen) return null;

  return (
    <aside
      role="complementary"
      aria-label="AI assistant panel"
      data-testid="ai-assistant-panel"
      className={clsx(

        'transition-transform duration-300 ease-in-out',
      )}
    >
      {/* Header */}
      <header className="flex items-center justify-between border-b border-gray-200 px-4 py-3">
        <div className="flex items-center gap-2">
          <Sparkles
            className="h-5 w-5 text-violet-500"
            aria-hidden="true"
          />
          <h2 className="text-sm font-semibold text-gray-900">
            AI Assistant
          </h2>
          <span className="rounded-full bg-violet-100 px-2 py-0.5 text-xs font-medium text-violet-700">
            Preview
          </span>
        </div>
        <button
          onClick={onClose}
          aria-label="Close AI assistant"
          className="rounded-md p-1 text-gray-400 hover:bg-gray-100 hover:text-gray-600 focus:outline-none focus:ring-2 focus:ring-violet-500"
        >
          <X className="h-4 w-4" />
        </button>
      </header>

      {/* Coming-soon notice */}
      <div className="flex items-start gap-2 bg-violet-50 px-4 py-3 text-xs text-violet-700">
        <Lock className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden="true" />
        <p>
          AI generation is not yet active. This preview shows where planning
          help will live.
        </p>
      </div>

      {/* Message thread */}
      <div
        className="flex-1 overflow-y-auto px-4 py-4 space-y-4"
        aria-live="polite"
        aria-label="Conversation"
      >
        {messages.map((msg) => (
          <MessageBubble key={msg.id} message={msg} />
        ))}
        {isLoading && <TypingIndicator />}

          <p className="mb-2 text-xs font-medium text-gray-500">
            Try asking…

            {SUGGESTED_PROMPTS.map((prompt) => (
